Rungs 1 and 2

Lab 1 · Worked example: trinomial American option pricing

The script evaluates an American put option on a recombining trinomial lattice with Boyle (1977) moment-matching probabilities, demonstrating optimal early exercise along the Snell envelope.

"""DV4 Project Rung 1: Worked example of trinomial lattice pricing for American options.

SIMULATED market data under geometric Brownian motion.
Prices an American put option on a 3-branch recombining trinomial lattice (Boyle 1977),
extracting the early exercise boundary and verifying the Snell envelope property.
"""

from __future__ import annotations

import math

import numpy as np


def run_worked_example() -> None:
    spot = 100.0
    strike = 100.0
    rate = 0.05
    vol = 0.20
    maturity = 1.0
    n_steps = 100

    dt = maturity / n_steps
    dx = vol * math.sqrt(3.0 * dt)
    u = math.exp(dx)
    1.0 / u
    df = math.exp(-rate * dt)
    nu = (rate - 0.5 * vol * vol) * dt

    pu = 1.0 / 6.0 + 0.5 * (nu * nu / (dx * dx) + nu / dx)
    pm = 2.0 / 3.0 - (nu * nu / (dx * dx))
    pd = 1.0 / 6.0 + 0.5 * (nu * nu / (dx * dx) - nu / dx)

    # Terminal node payoff
    indices = np.arange(-n_steps, n_steps + 1)
    s_terminal = spot * np.exp(indices * dx)
    v_amer = np.maximum(strike - s_terminal, 0.0)
    v_eur = v_amer.copy()

    early_exercise_count = 0

    for k in range(n_steps - 1, -1, -1):
        # Discounted expected continuation
        cont_amer = df * (pu * v_amer[2 : 2 * k + 3] + pm * v_amer[1 : 2 * k + 2] + pd * v_amer[0 : 2 * k + 1])
        cont_eur = df * (pu * v_eur[2 : 2 * k + 3] + pm * v_eur[1 : 2 * k + 2] + pd * v_eur[0 : 2 * k + 1])

        k_indices = np.arange(-k, k + 1)
        s_k = spot * np.exp(k_indices * dx)
        intrinsic = np.maximum(strike - s_k, 0.0)

        # American early exercise check (Snell envelope)
        exercise_now = intrinsic > cont_amer
        early_exercise_count += int(np.sum(exercise_now))

        v_amer = np.maximum(cont_amer, intrinsic)
        v_eur = cont_eur

    amer_price = float(v_amer[0])
    eur_price = float(v_eur[0])
    premium = amer_price - eur_price

    # Analytical BSM benchmark
    d1 = (math.log(spot / strike) + (rate + 0.5 * vol * vol) * maturity) / (vol * math.sqrt(maturity))
    d2 = d1 - vol * math.sqrt(maturity)
    n_neg_d1 = 0.5 * (1.0 + math.erf(-d1 / math.sqrt(2.0)))
    n_neg_d2 = 0.5 * (1.0 + math.erf(-d2 / math.sqrt(2.0)))
    bsm_eur = strike * math.exp(-rate * maturity) * n_neg_d2 - spot * n_neg_d1

    print(f"Trinomial American put price: {amer_price:.4f}")
    print(f"Trinomial European put price: {eur_price:.4f}")
    print(f"Analytical BSM European price: {bsm_eur:.4f}")
    print(f"American early-exercise premium: {premium:.4f}")
    print(f"Total early-exercise node events: {early_exercise_count}")


run_worked_example()

Lab 2 · Completion problem: Crank–Nicolson American PDE solver

Completing the Thomas tridiagonal forward sweep and Brennan–Schwartz obstacle condition yields an unconditionally stable second-order finite difference solution for the American put free-boundary problem.

"""DV4 Project Rung 2: Completion problem for Crank-Nicolson PDE American pricer.

SIMULATED market data under geometric Brownian motion.
Completes the Thomas algorithm forward sweep and the Brennan-Schwartz obstacle condition
enforcing the early-exercise free boundary on a log-spot spatial grid.
"""

from __future__ import annotations

import math

import numpy as np


def solve_american_pde(
    spot: float,
    strike: float,
    rate: float,
    vol: float,
    maturity: float,
    n_time: int = 100,
    n_spot: int = 100,
) -> float:
    std = vol * math.sqrt(maturity)
    x_max = 4.0 * std + rate * maturity
    x_min = -4.0 * std
    dx = (x_max - x_min) / (n_spot - 1)
    dt = maturity / n_time

    x = np.linspace(x_min, x_max, n_spot)
    s_grid = strike * np.exp(x)
    intrinsic = np.maximum(strike - s_grid, 0.0)
    v = intrinsic.copy()

    mu = rate - 0.5 * vol * vol
    alpha = 0.5 * (vol * vol) / (dx * dx)
    beta = 0.5 * mu / dx

    a_coef = alpha - beta
    b_coef = -2.0 * alpha - rate
    c_coef = alpha + beta

    theta = 0.5  # Crank-Nicolson
    m = n_spot - 2

    # Tridiagonal system matrix coefficients
    diag_a = -dt * theta * a_coef * np.ones(m)
    diag_b = (1.0 - dt * theta * b_coef) * np.ones(m)
    diag_c = -dt * theta * c_coef * np.ones(m)

    for step in range(1, n_time + 1):
        tau = step * dt
        rhs = v[1:-1] + dt * (1.0 - theta) * (a_coef * v[:-2] + b_coef * v[1:-1] + c_coef * v[2:])

        bc_lower = strike * math.exp(-rate * tau) - s_grid[0]
        bc_upper = 0.0

        rhs[0] += dt * theta * a_coef * bc_lower
        rhs[-1] += dt * theta * c_coef * bc_upper

        # Thomas algorithm forward sweep
        c_p = np.zeros(m)
        d_p = np.zeros(m)
        c_p[0] = diag_c[0] / diag_b[0]
        d_p[0] = rhs[0] / diag_b[0]

        for i in range(1, m):
            denom = diag_b[i] - diag_a[i] * c_p[i - 1]
            c_p[i] = diag_c[i] / denom if i < m - 1 else 0.0
            d_p[i] = (rhs[i] - diag_a[i] * d_p[i - 1]) / denom

        # Backward substitution with Brennan-Schwartz obstacle condition
        v_next = np.zeros(m)
        v_next[-1] = max(d_p[-1], intrinsic[-2])
        for i in range(m - 2, -1, -1):
            sol_val = d_p[i] - c_p[i] * v_next[i + 1]
            # Enforce American early exercise free boundary obstacle
            v_next[i] = max(___, ___)

        v[0] = bc_lower
        v[-1] = bc_upper
        v[1:-1] = v_next

    return float(np.interp(spot, s_grid, v))


def main() -> None:
    spot = 100.0
    strike = 100.0
    rate = 0.05
    vol = 0.20
    maturity = 1.0

    price = solve_american_pde(spot, strike, rate, vol, maturity, n_time=120, n_spot=120)
    print(f"Crank-Nicolson American put price: {price:.4f}")

Reading list

  1. Boyle (1977) — Options: A Monte Carlo approach, introducing simulation methods for contingent claims valuation.
  2. Cox, Ross & Rubinstein (1979) — Option pricing: A simplified approach, introducing discrete binomial recombining lattices.
  3. Black & Scholes (1973) — The Pricing of Options and Corporate Liabilities, deriving the continuous-time no-arbitrage PDE and closed-form European formula.
  4. Merton (1973) — Theory of Rational Option Pricing, extending continuous-time replication to dividends and American boundaries.
  5. Longstaff & Schwartz (2001) — Valuing American Options by Simulation: A Simple Least-Squares Approach, formulating LSMC conditional expectation regression.
  6. Carr & Madan (1999) — Option valuation using the fast Fourier transform, deriving damped characteristic function inversion.
  7. Fang & Oosterlee (2008) — A Novel Pricing Method for European Options Based on Fourier-Cosine Series Expansions, formulating the exponential-convergence COS algorithm.
  8. Glasserman (2003) — Monte Carlo Methods in Financial Engineering, comprehensive treatment of simulation, Greeks, and variance reduction.
