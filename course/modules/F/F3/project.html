    """Lab F3-project-1: Project Rung 1 Worked Example - Multi-Asset Factor Book.

Executes a complete multi-asset factor allocation across Rates, FX, and
Commodities, constructing Carry, Time-Series Momentum, and Value factors,
evaluating cross-asset correlations, crash skewness, and Sharpe ratio expansion
under an equal-risk parity allocation engine.
"""

from __future__ import annotations

import numpy as np
import scipy.stats


def compute_carry_factor(signals: np.ndarray, returns: np.ndarray) -> np.ndarray:
    """Build dollar-neutral cross-sectional carry factor returns."""
    t_obs, n_assets = signals.shape
    factor_rets = np.zeros(t_obs - 1)
    for t in range(t_obs - 1):
        sig = signals[t, :]
        demeaned = sig - np.mean(sig)
        denom = np.sum(np.abs(demeaned))
        weights = np.zeros(n_assets) if denom < 1e-12 else demeaned / denom
        factor_rets[t] = float(np.sum(weights * returns[t + 1, :]))
    return factor_rets


def compute_tsmom_factor(
    returns: np.ndarray,
    lookback: int = 12,
    target_vol: float = 0.10,
) -> np.ndarray:
    """Build time-series momentum factor with trailing inverse-volatility scaling."""
    t_obs, n_assets = returns.shape
    n_eval = t_obs - lookback
    factor_rets = np.zeros(n_eval)
    for idx, t in enumerate(range(lookback, t_obs)):
        past_perf = np.sum(returns[t - lookback : t, :], axis=0)
        direction = np.sign(past_perf)
        vol_win = max(0, t - 24)
        hist_vol = np.std(returns[vol_win:t, :], axis=0, ddof=1) * np.sqrt(12.0)
        hist_vol = np.maximum(hist_vol, 0.02)
        pos = direction * (target_vol / hist_vol) / float(n_assets)
        factor_rets[idx] = float(np.sum(pos * returns[t, :]))
    return factor_rets


def compute_value_factor(signals: np.ndarray, returns: np.ndarray) -> np.ndarray:
    """Build dollar-neutral cross-sectional value factor returns."""
    t_obs, n_assets = signals.shape
    factor_rets = np.zeros(t_obs - 1)
    for t in range(t_obs - 1):
        sig = signals[t, :]
        demeaned = sig - np.mean(sig)
        denom = np.sum(np.abs(demeaned))
        weights = np.zeros(n_assets) if denom < 1e-12 else demeaned / denom
        factor_rets[t] = float(np.sum(weights * returns[t + 1, :]))
    return factor_rets


def compute_factor_diagnostics(
    factor_returns: np.ndarray,
    annualization: float = 12.0,
) -> dict[str, float]:
    """Compute risk, return, and crash tail diagnostics for a factor return series."""
    mean_val = float(np.mean(factor_returns))
    vol_val = float(np.std(factor_returns, ddof=1))
    mean_ann = mean_val * annualization * 100.0
    vol_ann = vol_val * np.sqrt(annualization) * 100.0
    sharpe = float(mean_ann / vol_ann) if vol_ann > 1e-8 else 0.0
    skew = float(scipy.stats.skew(factor_returns, bias=False))
    kurt = float(scipy.stats.kurtosis(factor_returns, bias=False))
    cum_ret = np.cumsum(factor_returns)
    running_max = np.maximum.accumulate(cum_ret)
    drawdowns = cum_ret - running_max
    max_dd = float(np.min(drawdowns) * 100.0) if len(drawdowns) > 0 else 0.0
    worst_m = float(np.min(factor_returns) * 100.0)

    return {
        "mean_ann": round(mean_ann, 2),
        "vol_ann": round(vol_ann, 2),
        "sharpe": round(sharpe, 2),
        "skewness": round(skew, 2),
        "kurtosis": round(kurt, 2),
        "max_drawdown": round(max_dd, 2),
        "worst_month": round(worst_m, 2),
    }


def run_worked_example(seed: int = 42) -> dict[str, object]:
    """Execute the multi-asset factor book worked pipeline on synthetic multi-asset data."""
    rng = np.random.default_rng(seed)
    n_months = 120
    n_assets = 4

    classes = ["rates", "fx", "commodities"]
    factor_streams: dict[str, np.ndarray] = {}
    lookback = 12

    for ac in classes:
        rets = rng.normal(0.005, 0.03, size=(n_months, n_assets))
        carry_sig = rng.normal(0.02, 0.01, size=(n_months, n_assets))
        val_sig = rng.normal(0.0, 0.05, size=(n_months, n_assets))

        c_raw = compute_carry_factor(carry_sig, rets)
        m_raw = compute_tsmom_factor(rets, lookback=lookback)
        v_raw = compute_value_factor(val_sig, rets)

        start_idx = lookback - 1
        factor_streams[f"{ac}_carry"] = c_raw[start_idx:]
        factor_streams[f"{ac}_tsmom"] = m_raw
        factor_streams[f"{ac}_value"] = v_raw[start_idx:]

    factor_names = sorted(factor_streams.keys())
    matrix = np.column_stack([factor_streams[name] for name in factor_names])

    vols = np.std(matrix, axis=0, ddof=1)
    inv_vols = 1.0 / np.maximum(vols, 1e-6)
    weights = inv_vols / np.sum(inv_vols)

    book_returns = matrix @ weights
    book_diag = compute_factor_diagnostics(book_returns)
    factor_diags = {name: compute_factor_diagnostics(factor_streams[name]) for name in factor_names}
    mean_standalone_sharpe = float(np.mean([d["sharpe"] for d in factor_diags.values()]))
    book_sharpe = book_diag["sharpe"]
    expansion_ratio = round(float(book_sharpe / mean_standalone_sharpe), 2) if mean_standalone_sharpe > 0 else 0.0

    return {
        "factor_names": factor_names,
        "weights": {name: round(float(w), 4) for name, w in zip(factor_names, weights, strict=True)},
        "book_returns": book_returns,
        "book_diagnostics": book_diag,
        "factor_diagnostics": factor_diags,
        "mean_standalone_sharpe": round(mean_standalone_sharpe, 2),
        "expansion_ratio": expansion_ratio,
    }

Specification

Context
You are a quantitative macro factor portfolio manager at a global multi-asset asset manager. The Chief Investment Officer has commissioned a systematic multi-asset factor book spanning Sovereign Rates, Foreign Exchange, and Commodities. You are tasked with implementing dollar-neutral Carry and Value strategies, a volatility-scaled Time-Series Momentum (TSMOM) strategy, diagnostic routines to evaluate higher-order risk moments (skewness, kurtosis, peak-to-trough drawdowns), and an equal-risk-parity portfolio synthesis engine that generates robust cross-asset diversification.
Inputs
A deterministic instance dictionary generated by make_instance(seed) containing:
  • "rates": A dictionary containing "returns", "carry", and "value" arrays of shape \((360, 4)\) for 4 sovereign bond futures contracts over 360 months.
  • "fx": A dictionary containing "returns", "carry", and "value" arrays of shape \((360, 4)\) for 4 liquid G10 currency pairs.
  • "commodities": A dictionary containing "returns", "carry", and "value" arrays of shape \((360, 4)\) for 4 major commodity futures (energy, metals, agriculture).
  • "n_months": Integer (360) indicating total sample length.
  • "seed": Integer seed used for deterministic reproducibility.
Required Outputs
A Python module named factor_book.py implementing five callable functions:
  1. compute_carry_factor(signals: np.ndarray, returns: np.ndarray) -> np.ndarray: Computes dollar-neutral, cross-sectionally demeaned carry factor returns of length \(T - 1\).
  2. compute_tsmom_factor(returns: np.ndarray, lookback: int = 12, target_vol: float = 0.10) -> np.ndarray: Computes time-series momentum factor returns with trailing 24-month inverse-volatility scaling, of length \(T - \text{lookback}\).
  3. compute_value_factor(signals: np.ndarray, returns: np.ndarray) -> np.ndarray: Computes dollar-neutral, cross-sectionally demeaned value factor returns of length \(T - 1\).
  4. compute_factor_diagnostics(factor_returns: np.ndarray, annualization: float = 12.0) -> dict[str, float]: Computes annualized mean %, annualized volatility %, Sharpe ratio, sample skewness, sample kurtosis, peak-to-trough maximum drawdown %, and worst month %.
  5. construct_multi_asset_factor_book(instance_data: dict[str, object], weighting_scheme: str = "risk_parity") -> dict[str, object]: Integrates all 9 factor streams into a composite book under either "risk_parity" or "equal_weight", returning factor names, weights, composite returns, individual and book diagnostics, correlation matrix, mean standalone Sharpe, and diversification expansion ratio.
Constraints
All functions must strictly validate input shapes and types, raising ValueError on invalid inputs (mismatched shapes, sample lengths smaller than lookback horizons, invalid weighting schemes, or fewer than 2 observations for diagnostics). Weights in carry and value must sum to zero across assets at every time step (dollar neutrality). All operations must use vectorized NumPy or SciPy.

Exemplar 1 · Level 4 (Distinguished)

Executive Summary: Multi-Asset Systematic Factor Book Allocation

We present an institutional multi-asset factor book integrating Carry, Time-Series Momentum (TSMOM), and Value across Sovereign Rates, Foreign Exchange, and Commodities over 30 years of data. Constructing factor streams following Asness, Moskowitz & Pedersen (2013), Koijen, Moskowitz, Pedersen & Vrugt (2018), and Moskowitz, Ooi & Pedersen (2012) reveals compelling empirical properties. Standalone factor Sharpe ratios average \(0.53\), with individual factor streams displaying expected risk asymmetries: Carry across Rates, FX, and Commodities exhibits pronounced negative return skewness (averaging \(-0.88\)), reflecting exposure to liquidity and margin unwind spirals (Brunnermeier et al. 2008), while TSMOM delivers positive crisis alpha and convex smile payoffs (Fung & Hsieh 2001). Value and TSMOM display strong negative within-asset correlation (\(\rho = -0.48\)). By allocating across all 9 factor streams via an inverse-volatility risk-parity engine, the composite factor book achieves an annualized Sharpe ratio of \(1.15\)—a diversification expansion ratio of \(2.17\times\) over standalone components. We address the Huang, Li, Wang & Zhou (2020) pooled-regression critique by demonstrating that while between-asset mean dispersion inflates pooled panel OLS \(t\)-statistics, the multi-asset portfolio trading strategy remains highly profitable because investors earn the unconditional asset risk premia alongside trend-timing alpha. We recommend a full institutional risk-parity allocation to the multi-asset factor book.
