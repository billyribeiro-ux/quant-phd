Three statements from raw transactions

A small accounting team needs one library that posts a ledger of double-entry transactions and reports the balance sheet, the income statement and the cash flow statement for any month end, and that checks whether the three statements articulate. Rungs 1 and 2

Lab 1 · Worked example: posting ten transactions to the three statements

The code posts ten SIMULATED transactions, rejects one entry that does not balance, and then reports the balance sheet at two month ends and the income statement and cash flow statement for February. Amounts are integer cents. Before running it, predict the February net income. The February entries that touch revenue or expense accounts are one sale and its cost, in one transaction, and nothing else.

from __future__ import annotations

from typing import Any

# SIMULATED ledger: ten transactions for one company in January and February 2024 (not a real company).
# Every amount is an integer number of cents. Each line carries a debit or a credit, never both.
TRANSACTIONS: list[dict[str, Any]] = [

    {

def post(transactions: list[dict[str, Any]]) -> list[tuple[str, str, int]]:
    """Check each entry and return its postings as (date, account, debit minus credit), in cents."""
    postings: list[tuple[str, str, int]] = []
    for entry in transactions:
        total = 0
        for line in entry["lines"]:
            debit, credit = int(line["debit"]), int(line["credit"])
            if (debit > 0) == (credit > 0):
                raise ValueError("a line carries a debit or a credit, and exactly one")
            postings.append((entry["date"], line["account"], debit - credit))
            total += debit - credit
        if total != 0:
            raise ValueError("unbalanced entry: the debits do not equal the credits")
    return postings


def balances_through(postings: list[tuple[str, str, int]], end: str) -> dict[str, int]:
    """Debit minus credit for each account, over the postings dated on or before end (ISO dates compare)."""
    totals: dict[str, int] = {}
    for date, account, signed in postings:
        if date <= end:
            totals[account] = totals.get(account, 0) + signed
    return totals


def main() -> None:
    postings = post(TRANSACTIONS)
    print("postings:", len(postings), "from", len(TRANSACTIONS), "entries, each entry balances")
    try:
        post([{"date": "2024-03-01", "lines": [{"account": "cash", "debit": 100, "credit": 0}]}])
    except ValueError as error:
        print("rejected:", error)

    for end in ("2024-01-31", "2024-02-29"):
        bal = balances_through(postings, end)
        # Assets are debit balances, net of the credit balance in accumulated depreciation.
        cash = bal.get("cash", 0)
        receivables = bal.get("accounts_receivable", 0)
        inventory = bal.get("inventory", 0)
        net_equipment = bal.get("equipment", 0) + bal.get("accumulated_depreciation", 0)
        assets = cash + receivables + inventory + net_equipment
        # Liabilities and common stock are credit balances, so their amounts are negated.
        payables = -bal.get("accounts_payable", 0)
        common_stock = -bal.get("common_stock", 0)
        # Net income to date is revenue less expenses; dividends reduce retained earnings.
        income_to_date = -(
            bal.get("sales", 0)
            + bal.get("cost_of_goods_sold", 0)
            + bal.get("operating_expenses", 0)
            + bal.get("depreciation_expense", 0)
        )
        retained = income_to_date - bal.get("dividends", 0)
        gap = assets - (payables + common_stock + retained)
        print(
            f"balance sheet {end}: assets {assets} (cash {cash}, receivables {receivables}, "
            f"inventory {inventory}, net equipment {net_equipment})"
        )
        print(f"  liabilities {payables}, common stock {common_stock}, retained earnings {retained}, gap {gap}")

    # Income statement for February: the change in each flow account between the two month ends.
    jan = balances_through(postings, "2024-01-31")
    feb = balances_through(postings, "2024-02-29")
    flow = {account: feb.get(account, 0) - jan.get(account, 0) for account in feb}
    sales = -flow.get("sales", 0)
    gross = sales - flow.get("cost_of_goods_sold", 0)
    net_income = gross - flow.get("operating_expenses", 0) - flow.get("depreciation_expense", 0)
    print("income statement February: sales", sales, "gross profit", gross, "net income", net_income)

    # Cash flow statement for February, indirect method: each balance change sets its cash effect.
    def change(account: str) -> int:
        return feb.get(account, 0) - jan.get(account, 0)

    operating = (
        net_income
        + flow.get("depreciation_expense", 0)
        - change("accounts_receivable")
        - change("inventory")
        - change("accounts_payable")
    )
    investing = -change("equipment")
    financing = -change("long_term_debt") - change("common_stock") - change("dividends")
    beginning = jan.get("cash", 0)
    ending = beginning + operating + investing + financing
    print(
        "cash flow February: operating",
        operating,
        "investing",
        investing,
        "financing",
        financing,
        "ending cash",
        ending,
    )
    print("ending cash equals the February balance sheet cash:", ending == feb.get("cash", 0))

Rung 3 · Specification

Context
A small accounting team wants one library that a later analysis can call. It takes a ledger of double-entry transactions, posts them, and reports for any month end the balance sheet, the income statement and the cash flow statement, and an articulation check that tests whether the three statements agree with one another. You implement the library to the specification, and the tests check that it reports the statements the way the lessons define them: balances through the date, the flows of the calendar month, and cash flows built from net income (the indirect method, defined under Conventions below). The ledgers in the labs and the autograder's instance are labelled SIMULATED. The autograder's ledger is generated from the workspace seed; the labs' ledgers are written out by hand in the lab code.
Inputs
A transaction is a mapping with a date in the form YYYY-MM-DD and a list of lines. A line is a mapping with an account, a debit and a credit. The amounts are integer cents: a line carries a debit or a credit, and exactly one of them is positive. Each transaction's debits must equal its credits. The chart of accounts, ACCOUNTS, is given in the starter with the class of each of its thirteen accounts. The autograder's ledger comes from grader/instance.py: make_instance(seed) returns a dict whose payload is a JSON string of 200 transactions dated in 2024. A student does not call make_instance; the grader decodes the payload. Every instance is labelled SIMULATED: it is generated, it is not a real company, and nothing in it is a real company's figure.
Required outputs
A file statements.py that defines ACCOUNTS, as given in the starter, and these five functions with these names and parameters:
  • post(transactions) returns a ledger. The ledger's type is yours to choose; callers pass it to the other four functions and never look inside it.
  • balance_sheet(ledger, period_end) returns a dict with the keys cash, accounts_receivable, inventory, equipment, accumulated_depreciation, net_equipment, total_assets, accounts_payable, long_term_debt, total_liabilities, common_stock, retained_earnings, total_equity and total_liabilities_and_equity.
  • income_statement(ledger, period_end) returns a dict with the keys sales, cost_of_goods_sold, gross_profit, operating_expenses, depreciation_expense and net_income.
  • cash_flow_statement(ledger, period_end) returns a dict with the keys net_income, depreciation, receivables, inventory, payables, cash_from_operations, capital_expenditure, cash_from_investing, net_borrowing, stock_issued, dividends_paid, cash_from_financing, net_change_in_cash, beginning_cash and ending_cash.
  • articulation_check(ledger, period_end) returns a dict with the keys balance_gap, cash_gap, retained_earnings_gap and depreciation_gap.
Every value in every returned dict is an int in cents. Start from the starter file: every function raises NotImplementedError until you replace it, and the suite fails on the starter.
Constraints
The standard library and nothing else. The module must not import any file from the grader, so it runs on its own. Functions validate their inputs. The three statement functions never print, post never changes its argument, and balance_sheet returns a new dict on each call. Every named error condition raises ValueError; the grader checks the exception type and not the wording of the message. The named conditions are these. post raises for a transaction without a date or without lines; a date that is not in the form YYYY-MM-DD or is not a real calendar date; an empty list of lines; a line without an account, a debit or a credit; an account outside the chart; an amount that is not an integer, including a boolean or a float such as 100.0; a negative amount; a line with both a debit and a credit, or with neither; and an entry whose debits do not equal its credits. Each of the three statement functions and the articulation check raises for a period end that is not a month end, which means the last day of a calendar month in the form YYYY-MM-DD. Money is never converted and never adjusted: the engine reports the cents it is given.
Grading mode
Autograded in your browser. grader/test_project.py collects 264 checks, counted with pytest --collect-only on the reference solution, and the count is 264 for each of the seeds 0 to 49. It runs with warnings as errors and covers: the chart of accounts; the standard-library imports; the posting rules and each named error, with their boundary cases (an entry that balances only with another entry, a one-cent entry, a zero on one side of a line, a repeated entry, two entries on one date, an empty list of transactions); the statement functions' silence and their return values; the balance sheet, income statement and cash flow statement against expected values computed from the seeded ledger, for each of the twelve 2024 month ends; the balance identity (assets equal liabilities plus equity); the cash tie (the ending cash equals the balance sheet cash); the net change in cash against the cash lines of each month; retained earnings against net income less dividends; the operating cash flow formula; the four articulation gaps; the month-end rule; and a hand-worked January ledger with values given on paper. A failing check names the property that failed and never prints the expected value. The suite is not secret. The ledger changes with the seed, and the expected values change with it.

Conventions, in brief

  • Balance through a date. For each account, the sum of debit minus credit over the entries dated on or before that date. Assets other than accumulated depreciation have a normal debit balance, and so do dividends and expenses. The other classes, accumulated depreciation included, have credit balances, and the statements show those credit balances as positive amounts.
  • The month. The period that ends on period_end is the calendar month that ends that day. Its previous month end is the last day of the month before. The flow of an account over the month is its balance through period_end less its balance through the previous month end.
  • Balance sheet. Assets are the debit balances, with accumulated depreciation shown as a positive amount and subtracted to give net_equipment. Liabilities and common stock are shown as positive amounts. Retained earnings are the net income to date less the dividends to date, where net income to date is revenue less cost of goods sold, operating expenses and depreciation.
  • Income statement. Sales are the negative of the sales flow. Cost, expense and depreciation flows are shown as positive amounts. Gross profit is sales less cost of goods sold, and net income is gross profit less operating expenses and depreciation.
  • Cash flow statement, indirect method. depreciation is the depreciation expense of the month. receivables and inventory are the negatives of their increases, and payables is the increase in accounts payable. cash_from_operations is net income plus depreciation plus those three terms, which are the cash effects of the changes in receivables, inventory and payables. capital_expenditure is the negative of the increase in equipment. net_borrowing is the increase in long-term debt, stock_issued the increase in common stock, and dividends_paid the negative of the increase in the dividends account. beginning_cash is the cash balance through the previous month end, and ending_cash is the beginning cash plus the net change in cash.
  • Articulation gaps. balance_gap is total assets less total liabilities and equity. It is zero for every ledger that post accepts, because post accepts only entries whose debits equal their credits, so the balances of all accounts sum to zero; the suite checks that it is zero. The suite does not test the sign of this always-zero quantity, since a reversed sign also gives zero.
  • cash_gap is the balance sheet cash less the ending cash. retained_earnings_gap is the change in retained earnings less the sum of net income and dividends_paid. depreciation_gap is the change in the accumulated depreciation shown on the balance sheet, as a positive amount, less the depreciation expense of the month. A nonzero depreciation_gap means the month's net credits to accumulated depreciation differ from its depreciation expense.
  • Concepts used here. The double-entry rule is defined in Lesson 1, the three statements in Lesson 2, and the articulation identities in Lesson 3. The conventions above apply those definitions.

    """Net income to date: revenue less cost of goods sold, operating expenses and depreciation."""
    return -(
        bal.get("sales", 0)
        + bal.get("cost_of_goods_sold", 0)
        + bal.get("operating_expenses", 0)
        + bal.get("depreciation_expense", 0)
    )

    mar = balance_through("2024-03-31")

    def change(account: str) -> int:
        return mar.get(account, 0) - feb.get(account, 0)

    # Balance sheet at 2024-03-31: assets are debit balances, less the credit in accumulated depreciation.
    cash = mar.get("cash", 0)
    net_equipment = mar.get("equipment", 0) + mar.get("accumulated_depreciation", 0)
    inventory = mar.get("inventory", 0)
    assets = cash + inventory + net_equipment
    liabilities = -(mar.get("accounts_payable", 0) + mar.get("long_term_debt", 0))
    common_stock = -mar.get("common_stock", 0)

        "liabilities",
        liabilities,
        "common stock",
        common_stock,
        "retained earnings",
        retained,
    )

    # Income statement for March: the change in each flow account from 2024-02-29 to 2024-03-31.
    net_income = -(
        change("sales") + change("cost_of_goods_sold") + change("operating_expenses") + change("depreciation_expense")
    )

    # Cash flow statement for March, indirect method.
    operating = (
        net_income

        - change("accounts_receivable")
        - change("inventory")
