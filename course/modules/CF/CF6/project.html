"""CF6 Project Lab 1 (Rung 1 Worked Example): Full IPO Pricing and Syndicate Model.

Holdings to BUILDER_RULES §6: deterministic, self-contained, standard library.
Demonstrates end-to-end execution of the investment banking capital raising model.
"""

from __future__ import annotations

import math
from decimal import ROUND_HALF_UP, Decimal
from typing import Any


def rock_equilibrium_price(
    v_low: float,
    v_high: float,
    prob_high: float,
    s_shares: int,
    informed_demand: int,
    uninformed_budget: float,
) -> tuple[float, float, float]:
    e_v = (1.0 - prob_high) * v_low + prob_high * v_high

    def expected_profit(p: float) -> float:
        d_u = uninformed_budget / p
        alloc_l = min(d_u, float(s_shares))
        tot_high_dem = d_u + float(informed_demand)
        alloc_h = float(s_shares) / tot_high_dem * d_u if tot_high_dem > float(s_shares) else d_u
        return (1.0 - prob_high) * alloc_l * (v_low - p) + prob_high * alloc_h * (v_high - p)

    lo, hi = v_low, e_v
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if expected_profit(mid) > 0.0:
            lo = mid
        else:
            hi = mid

    p_star = 0.5 * (lo + hi)
    d_u_star = uninformed_budget / p_star
    total_demand_high = d_u_star + float(informed_demand)
    alpha_h = min(1.0, float(s_shares) / total_demand_high)
    underpricing = (e_v - p_star) / p_star
    return p_star, alpha_h, underpricing


def bookbuilding_allocation(
    bids: list[dict[str, Any]],
    total_shares: int,
    clearing_price: float,
    rule: str = "discretionary",
) -> dict[str, Any]:
    alloc: dict[str, int] = {b["bidder_id"]: 0 for b in bids}
    qualifying = [b for b in bids if b["bid_price"] >= clearing_price]
    total_qualifying_demand = sum(b["bid_shares"] for b in qualifying)

    if total_qualifying_demand <= total_shares:
        for b in qualifying:
            alloc[b["bidder_id"]] = b["bid_shares"]
    elif rule == "pro_rata":
        ratio = total_shares / total_qualifying_demand
        raw = {b["bidder_id"]: b["bid_shares"] * ratio for b in qualifying}
        base = {b["bidder_id"]: math.floor(raw[b["bidder_id"]]) for b in qualifying}
        rem = total_shares - sum(base.values())
        sorted_by_frac = sorted(
            qualifying,
            key=lambda b: raw[b["bidder_id"]] - base[b["bidder_id"]],
            reverse=True,
        )
        for i in range(rem):
            base[sorted_by_frac[i]["bidder_id"]] += 1
        alloc.update(base)
    elif rule == "discretionary":
        pos_bids = [b for b in qualifying if b.get("revealed_info") == "positive"]
        pos_demand = sum(b["bid_shares"] for b in pos_bids)
        if pos_demand <= total_shares:
            for b in pos_bids:
                alloc[b["bidder_id"]] = b["bid_shares"]
            rem_shares = total_shares - pos_demand
            other_bids = [b for b in qualifying if b.get("revealed_info") != "positive"]
            other_demand = sum(b["bid_shares"] for b in other_bids)
            if other_demand > 0 and rem_shares > 0:
                ratio = rem_shares / other_demand
                raw = {b["bidder_id"]: b["bid_shares"] * ratio for b in other_bids}
                base = {b["bidder_id"]: math.floor(raw[b["bidder_id"]]) for b in other_bids}
                rem = rem_shares - sum(base.values())
                sorted_others = sorted(
                    other_bids,
                    key=lambda b: raw[b["bidder_id"]] - base[b["bidder_id"]],
                    reverse=True,
                )
                for i in range(rem):
                    base[sorted_others[i]["bidder_id"]] += 1
                alloc.update(base)
        else:
            ratio = total_shares / pos_demand
            raw = {b["bidder_id"]: b["bid_shares"] * ratio for b in pos_bids}
            base = {b["bidder_id"]: math.floor(raw[b["bidder_id"]]) for b in pos_bids}
            rem = total_shares - sum(base.values())
            sorted_pos = sorted(
                pos_bids,
                key=lambda b: raw[b["bidder_id"]] - base[b["bidder_id"]],
                reverse=True,
            )
            for i in range(rem):
                base[sorted_pos[i]["bidder_id"]] += 1
            alloc.update(base)

    tot_alloc = sum(alloc.values())
    inst_shares = sum(alloc[b["bidder_id"]] for b in bids if b.get("investor_type") == "institutional")
    inst_frac = inst_shares / tot_alloc if tot_alloc > 0 else 0.0
    fulfill = {b["bidder_id"]: alloc[b["bidder_id"]] / b["bid_shares"] if b["bid_shares"] > 0 else 0.0 for b in bids}
    price_dec = Decimal(str(clearing_price))
    rev_cents = int((Decimal(tot_alloc) * price_dec * Decimal("100")).quantize(Decimal("1"), rounding=ROUND_HALF_UP))

    return {
        "allocations": alloc,
        "total_allocated": tot_alloc,
        "total_revenue_cents": rev_cents,
        "institutional_share_fraction": inst_frac,
        "fulfillment_rates": fulfill,
    }


def syndicate_economics(
    gross_proceeds_cents: int,
    spread_bps: int = 700,
    lead_manager_share: float = 0.60,
    co_manager_share: float = 0.30,
    selling_group_share: float = 0.10,
) -> dict[str, int]:
    spread_dec = Decimal(spread_bps) / Decimal(10000)
    total_spread = int((Decimal(gross_proceeds_cents) * spread_dec).quantize(Decimal("1"), rounding=ROUND_HALF_UP))
    net_proceeds = gross_proceeds_cents - total_spread

    lead_dec = Decimal(str(lead_manager_share))
    co_dec = Decimal(str(co_manager_share))
    lead_cents = int((Decimal(total_spread) * lead_dec).quantize(Decimal("1"), rounding=ROUND_HALF_UP))
    co_cents = int((Decimal(total_spread) * co_dec).quantize(Decimal("1"), rounding=ROUND_HALF_UP))
    selling_cents = total_spread - lead_cents - co_cents

    mgmt_fee = int((Decimal(total_spread) * Decimal("0.20")).quantize(Decimal("1"), rounding=ROUND_HALF_UP))
    uw_fee = int((Decimal(total_spread) * Decimal("0.20")).quantize(Decimal("1"), rounding=ROUND_HALF_UP))
    selling_concession = total_spread - mgmt_fee - uw_fee

    return {
        "gross_proceeds_cents": gross_proceeds_cents,
        "total_spread_cents": total_spread,
        "net_proceeds_cents": net_proceeds,
        "lead_manager_cents": lead_cents,
        "co_manager_cents": co_cents,
        "selling_group_cents": selling_cents,
        "management_fee_cents": mgmt_fee,
        "underwriting_fee_cents": uw_fee,
        "selling_concession_cents": selling_concession,
    }

Rung 3 · Specification

Context
An investment banking equity underwriting committee requires an algorithmic pricing and syndicate allocation engine (ipo_pricing.py) and an executive transaction memorandum (memo.md) advising an issuing client on their initial public offering and seasoned financing strategy. The engine solves the Rock (1986) winner's curse equilibrium, allocates shares across an institutional order book, evaluates Myers and Majluf (1984) adverse selection for follow-on offerings, measures long-run 36-month BHAR performance against matched benchmarks, and calculates exact underwriting fee waterfalls.
Inputs
The estimation engine ingests a parameter dictionary generated deterministically by make_instance(seed: int):
  1. Rock parameters: Low-state value rock_v_low, high-state value rock_v_high, high-state probability rock_prob_high, total shares rock_shares, informed demand rock_informed_demand, and uninformed budget rock_uninformed_budget.
  2. Bookbuilding order book: List of bid dictionaries bookbuilding_bids containing bidder_id, bid_price, bid_shares, and revealed_info, along with total_shares_offered and clearing_price.
  3. Myers-Majluf SEO parameters: Good asset value seo_v_good, bad asset value seo_v_bad, prior probability seo_prob_good, project outlay seo_project_cost, project NPV seo_project_npv, and internal cash slack seo_slack_cash.
  4. Long-run return streams: Monthly return lists for issuer (monthly_issuer_returns) and matched peer benchmark (monthly_benchmark_returns) across 36 months.
  5. Syndicate parameters: Gross offering proceeds in integer cents (gross_proceeds_cents), gross spread in basis points (spread_bps), and syndicate participant shares (lead_manager_share, co_manager_share, selling_group_share).
Required outputs
Two core deliverables:
  • ipo_pricing.py: Standard-library Python module implementing the five required functions: rock_equilibrium_price, bookbuilding_allocation, seo_announcement_impact, long_run_bhar, and syndicate_economics.
  • memo.md: Executive syndicate case memorandum (1–2 pages) presenting optimal offer pricing, allocation justifications, adverse selection mitigation, long-run abnormal return caveats, and exact syndicate compensation.
Constraints
  • Standard library only: no external dependencies (numpy, scipy). Use math and decimal.Decimal.
  • Monetary fee amounts and dollar balances must be calculated in integer cents with ROUND_HALF_UP precision, strictly conserving gross proceeds and total spread.
  • Share allocations must be non-negative integers summing exactly to total shares offered, using Hamilton-Hare largest-remainder rounding for pro-rata rationing.
  • All invalid inputs (e.g. non-positive share counts, budgets, outlays, out-of-bound probabilities, negative spreads, or unequal return series lengths) must raise ValueError.
Grading
Grading: both autograder and rubric. The autograder runs test_project.py across 50 pseudo-random instance seeds to verify mathematical convergence, share conservation, and corner case handling. The rubric evaluates the executive syndicate case memo across five analytical criteria.

1 Rock Winner's Curse & Equilibrium Pricing

Under a SIMULATED offering regime (Seed 42), the issuer faces asset outcomes \(V_L = \$12.50\) and \(V_H = \$26.00\) with prior \(\mathbb{P}(V_H) = 0.55\), offering \(S = 2,000,000\) shares. Informed investors demand 2,200,000 shares when \(V = V_H\), while uninformed retail investors commit a fixed budget of $45,000,000. In the low state, informed investors withdraw, allocating uninformed investors \(\min(D_U, S) = S\) shares. In the high state, total demand is \(D_U + 2,200,000\), rationing uninformed buyers to fraction \(\alpha_H = S / (D_U + 2,200,000) \approx 0.442\). Solving the zero-profit condition via bisection yields equilibrium offer price \(P^* = \$21.28\), implying an expected underpricing of 15.1% relative to unconditional expected value \(\bar{V} = \$20.23\). Underpricing is not an underwriting failure but an equilibrium necessity required to induce uninformed capital participation (rock1986new).

2 Bookbuilding Allocation & Information Extraction

We evaluate an institutional book of 8 qualifying bids totaling 3,250,000 shares against 2,000,000 shares offered at a clearing price of $21.50. Under Benveniste and Spindt (1989, reviewed in ljungqvist2007ipo), the bookrunner allocates 100% of demand to institutional investors revealing positive valuation signals (1,400,000 shares), rationing remaining bids (1,850,000 shares) across the 600,000 residual share pool. Using Hamilton-Hare largest-remainder integer rounding, all fractional shares are allocated deterministically to the highest decimal remainders, ensuring that \(\sum \text{allocated}_i \equiv 2,000,000\) exactly. This discretionary favoritism rewards information discovery, deterring institutional bid shading.

C2, Level 4: Clear application of Benveniste-Spindt discretionary rule, Hamilton-Hare largest remainder algorithm, and exact share conservation.

3 Myers-Majluf SEO Adverse Selection & Slack

For the firm's planned seasoned offering, existing assets-in-place are \(A_H = \$180\text{M}\) or \(A_L = \$95\text{M}\) (\(p = 0.50\)), requiring \(I = \$35\text{M}\) for an expansion project with NPV \(B = \$5.0\text{M}\). With zero internal cash slack (\(S = 0\)), the adverse selection dilution penalty is \(\Delta_{\text{dilution}} = \frac{35 \times 0.50 \times 85}{137.5 + 35 + 5} = \frac{1487.5}{177.5} \approx \$8.38\text{M}\). Because \(B = \$5.0\text{M} < \$8.38\text{M}\), management rejects the positive-NPV project (myers1984corporate). Holding internal cash slack of \(S \ge \$14.1\text{M}\) reduces required external equity to \(E \le \$20.9\text{M}\), lowering dilution below \$5.0M and eliminating underinvestment.

4 Long-Run Performance & Wealth Relative

Across 36 post-issue months, the issuer compounded to +12.4%, while its size-and-book-to-market matched non-issuing peer benchmark compounded to +48.6%. The Buy-and-Hold Abnormal Return is \(\text{BHAR} = 12.4\% - 48.6\% = -36.2\%\), and the Ritter wealth relative is \(WR = 1.124 / 1.486 = 0.7564\) (ritter1991longrun; loughran1995new). Cumulative abnormal return (CAR) measured \(-24.8\%\), diverging from BHAR due to monthly portfolio rebalancing bias and variance drag. We note Fama's (1998) joint-hypothesis caveat: long-horizon abnormal underperformance may reflect misestimated factor loadings on small-growth risk rather than market inefficiency.

Rung 4 · Stretch

In modern equity capital markets underwriting, investment banks manage aftermarket price volatility through the Greenshoe overallotment option (typically 15% of the base offering size). Under Rule 104 of SEC Regulation M, the lead bookrunner oversells the offering by 15%, establishing a syndicate short position. If the stock trades below the offer price in the immediate aftermarket, the bookrunner stabilizes the price by buying shares on the open exchange, covering the short at a profit while supporting the market. If the stock trades above the offer price, the bookrunner exercises the Greenshoe option at the fixed offer price, delivering primary shares without price risk.
