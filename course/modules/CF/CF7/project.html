"""Lab CF7-project-1: Project Rung 1 Worked Example.

Fully worked baseline model: historical Year 0 statements, 3-period projection,
closed-form revolver circularity resolution, DCF valuation, and integrity audit.
"""

from __future__ import annotations


def run_worked_example() -> dict[str, float]:
    """Execute complete worked example calculation for AeroParts Inc."""
    # Historical base
    rev_0 = 400.0
    cash_0 = 30.0
    net_debt_0 = 70.0
    c_min = 25.0
    r_prev = 10.0
    term_debt_0 = 90.0

    # Period 1 assumptions
    growth = 0.08
    gross_margin = 0.45
    sga_pct = 0.20
    depr_pct = 0.10
    capex_pct = 0.06
    dso = 45.0
    dio = 55.0
    dpo = 40.0
    tax_rate = 0.25
    term_rate = 0.06
    revolver_rate = 0.075
    cash_rate = 0.02
    term_amort = 15.0
    div_rate = 0.20
    wacc = 0.09
    terminal_g = 0.025
    shares = 10.0

    # 1. Operating Statement
    rev = rev_0 * (1.0 + growth)  # 432.0
    gp = rev * gross_margin  # 194.4
    cogs = rev - gp  # 237.6
    sga = rev * sga_pct  # 86.4
    ebitda = gp - sga  # 108.0
    net_ppe_0 = 180.0
    capex = rev * capex_pct  # 25.92
    depr = net_ppe_0 * depr_pct  # 18.0
    net_ppe_0 + capex - depr  # 187.92
    ebt_op = ebitda - depr  # 90.0

    # 2. Working Capital
    ar = rev * (dso / 365.0)  # 53.2603
    inv = cogs * (dio / 365.0)  # 35.7973
    ar_0 = rev_0 * (dso / 365.0)  # 49.3151
    inv_0 = (rev_0 * (1 - gross_margin)) * (dio / 365.0)  # 33.1507
    purchases = cogs + (inv - inv_0)
    ap = purchases * (dpo / 365.0)
    ap_0 = (rev_0 * (1 - gross_margin)) * (dpo / 365.0)

    delta_ar = ar - ar_0
    delta_inv = inv - inv_0
    delta_ap = ap - ap_0
    delta_nwc = delta_ar + delta_inv - delta_ap

    # 3. Term debt
    term_end = term_debt_0 - term_amort  # 75.0
    term_avg = 0.5 * (term_debt_0 + term_end)
    term_interest = term_avg * term_rate  # 82.5 * 0.06 = 4.95

    # 4. Revolver Circularity
    alpha = (1.0 - div_rate) * (1.0 - tax_rate)
    cf_core = alpha * (ebt_op - term_interest) + depr - delta_nwc - capex - term_amort
    c_bar_min = 0.5 * (cash_0 + c_min)
    num_r = (c_min - cash_0) - cf_core - alpha * cash_rate * c_bar_min + r_prev * (1.0 + 0.5 * alpha * revolver_rate)
    den_r = 1.0 - 0.5 * alpha * revolver_rate
    r_end = num_r / den_r

    r_avg = 0.5 * (r_prev + r_end)
    c_avg = 0.5 * (cash_0 + c_min)
    net_int = term_interest + r_avg * revolver_rate - c_avg * cash_rate

    ebt = ebt_op - net_int
    tax = ebt * tax_rate
    ni = ebt - tax
    ni * div_rate

    # 5. DCF
    nopat = ebt_op * (1.0 - tax_rate)
    fcff_1 = nopat + depr - capex - delta_nwc
    fcff_terminal = fcff_1 * (1.0 + terminal_g)
    tv = fcff_terminal / (wacc - terminal_g)
    pv_tv = tv / (1.0 + wacc)
    pv_fcff = fcff_1 / (1.0 + wacc)
    ev = pv_fcff + pv_tv
    equity_val = ev - net_debt_0
    eq_per_share = equity_val / shares

    return {
        "revenue": rev,
        "ebitda": ebitda,
        "net_income": ni,
        "revolver_end": r_end,
        "fcff_1": fcff_1,
        "enterprise_value": ev,
        "equity_per_share": eq_per_share,
    }


def main() -> None:
    res = run_worked_example()
    print("Projected Revenue:", f"{res['revenue']:.4f}")
    print("Projected EBITDA:", f"{res['ebitda']:.4f}")
    print("Projected Net Income:", f"{res['net_income']:.4f}")
    print("Ending Revolver Balance:", f"{res['revolver_end']:.4f}")
    print("Period 1 FCFF:", f"{res['fcff_1']:.4f}")
    print("Enterprise Value:", f"{res['enterprise_value']:.4f}")
    print("Equity Value Per Share:", f"{res['equity_per_share']:.4f}")

Rung 3 · Independent Project Specification

Context
You are advising a private equity sponsor evaluating a leveraged buyout of a mid-market manufacturing firm. The deal requires a dynamic three-statement projection model with a revolving credit line, debt amortization, and a discounted cash flow valuation. To guarantee institutional model governance, the investment committee mandates an independent Python reconciliation engine to prove that the spreadsheet workbook contains zero formula truncation, cell omission, or circular convergence errors.
Inputs
The problem instance is generated deterministically by grader/instance.py via make_instance(seed: int). The instance dictionary contains: company_name, currency, horizon (3 years), historical_data (Year 0 balance sheet and income statement actuals), operating_assumptions (revenue growth, gross margin, SG&A rate, CapEx rate, depreciation rate, DSO, DIO, DPO, tax rate), debt_assumptions (term amortization schedule, term interest rate, revolver commitment, revolver interest rate, cash interest rate, minimum cash, dividend payout ratio), and valuation_assumptions (WACC, terminal growth rate, shares outstanding).
Required Outputs
Two deliverables:
  1. model_reconciler.py implementing:
    • validate_inputs: validates parameter domain constraints, raising ValueError on violations.
    • solve_revolver_circularity: exact closed-form circularity solver returning revolver balance and cash.
    • build_model: full three-statement projection returning model schedules, valuation, and integrity.
    • compute_sensitivity_table: 2D valuation matrix.
    • reconcile_sheets: automated reconciliation auditor.
    • check_hardcodes: formula scanner flagging hard-coded numbers.
  2. A FAST-compliant workbook (.xlsx) matching the structure and outputs of your seeded case, accompanied by a written reconciliation audit report scored by the rubric.
Constraints
  • The balance sheet must balance across all periods with maximum error under \(10^{-4}\).
  • Cash flow statement ending cash must strictly match balance sheet cash.
  • Calculation cells must contain zero hard-coded numeric literals other than standard constants.
  • No plug accounts are permitted; all schedules articulate via double-entry flows.
  • Revolver interest circularity must be resolved dynamically in Python.
Grading Mode
Graded under both autograder and rubric modes. The Python reference suite tests the code implementation across 50 seeded instances in Pyodide. The workbook and reconciliation report are evaluated against the 5-criterion observable rubric.

def run_benchmark_audit() -> dict[str, float | bool]:
    """Execute integrity and reconciliation diagnostics for the benchmark case."""
    # Benchmark case inputs
    rev_0 = 500.0
    cash_0 = 40.0
    ar_0 = 60.0
    inv_0 = 45.0
    ppe_0 = 220.0
    ap_0 = 40.0
    term_debt_0 = 120.0
    revolver_0 = 10.0
    common_stock = 150.0
    re_0 = (cash_0 + ar_0 + inv_0 + ppe_0) - (ap_0 + term_debt_0 + revolver_0) - common_stock

    growth = 0.07
    gross_margin = 0.48
    sga_pct = 0.22
    depr_pct = 0.10
    capex_pct = 0.05
    dso = 45.0
    dio = 55.0
    dpo = 42.0
    tax_rate = 0.25
    term_rate = 0.065
    revolver_rate = 0.080
    cash_rate = 0.020
    c_min = 30.0
    term_amort = 24.0
    div_payout = 0.20
    wacc = 0.095
    terminal_g = 0.025
    shares = 20.0

    # Period 1 projection
    rev_1 = rev_0 * (1.0 + growth)
    gp_1 = rev_1 * gross_margin
    cogs_1 = rev_1 - gp_1
    sga_1 = rev_1 * sga_pct
    ebitda_1 = gp_1 - sga_1
    capex_1 = rev_1 * capex_pct
    depr_1 = ppe_0 * depr_pct
    net_ppe_1 = ppe_0 + capex_1 - depr_1
    ebt_op_1 = ebitda_1 - depr_1

    ar_1 = rev_1 * (dso / 365.0)
    inv_1 = cogs_1 * (dio / 365.0)
    purchases_1 = cogs_1 + (inv_1 - inv_0)
    ap_1 = purchases_1 * (dpo / 365.0)

    delta_ar = ar_1 - ar_0
    delta_inv = inv_1 - inv_0
    delta_ap = ap_1 - ap_0
    delta_nwc = delta_ar + delta_inv - delta_ap

    term_end_1 = term_debt_0 - term_amort
    term_avg_1 = 0.5 * (term_debt_0 + term_end_1)
    term_int_1 = term_avg_1 * term_rate

    alpha = (1.0 - div_payout) * (1.0 - tax_rate)
    cf_core_1 = alpha * (ebt_op_1 - term_int_1) + depr_1 - delta_nwc - capex_1 - term_amort

    # Branch 1: Surplus test with r_1 = 0.0
    num_c_zero = cash_0 * (1.0 + 0.5 * alpha * cash_rate) + cf_core_1 - revolver_0 * (1.0 + 0.5 * alpha * revolver_rate)
    den_c_zero = 1.0 - 0.5 * alpha * cash_rate
    c_zero = num_c_zero / den_c_zero

    if c_zero >= c_min:
        r_1 = 0.0
        cash_1 = c_zero
        delta_r = -revolver_0
    else:
        c_bar_min = 0.5 * (cash_0 + c_min)
        num_r = (
            (c_min - cash_0)
            - cf_core_1
            - alpha * cash_rate * c_bar_min
            + revolver_0 * (1.0 + 0.5 * alpha * revolver_rate)
        )
        den_r = 1.0 - 0.5 * alpha * revolver_rate
        r_1 = num_r / den_r
        cash_1 = c_min
        delta_r = r_1 - revolver_0

    r_avg_1 = 0.5 * (revolver_0 + r_1)
    cash_avg_1 = 0.5 * (cash_0 + cash_1)
    net_int_1 = term_int_1 + r_avg_1 * revolver_rate - cash_avg_1 * cash_rate

    ebt_1 = ebt_op_1 - net_int_1
    tax_1 = ebt_1 * tax_rate
    ni_1 = ebt_1 - tax_1
    div_1 = ni_1 * div_payout

    re_1 = re_0 + ni_1 - div_1
    equity_1 = common_stock + re_1
    assets_1 = cash_1 + ar_1 + inv_1 + net_ppe_1
    liab_1 = ap_1 + term_end_1 + r_1
