Lab 1 · Worked Example: 3-Factor Gaussian ATSM Kalman Filter Pipeline

Executing the Kalman filter recursion on 60 monthly observations of a 6-tenor yield curve with 5 bps measurement error recovers the latent state trajectory with factor correlations exceeding 0.90 and keeps overall cross-sectional pricing RMSE below 5 basis points.

"""Lab FI2-project-1: 3-Factor Gaussian ATSM Kalman filter worked example (Rung 1).

SIMULATED: end-to-end implementation of the 3-factor Gaussian Affine Term-Structure Model
Kalman filter pipeline on a simulated curve of monthly yields over 60 periods across 6 tenors.
"""

from __future__ import annotations

import numpy as np


def solve_loadings(
    tau: np.ndarray,
    delta0: float,
    kappa: np.ndarray,
    sigma: np.ndarray,
) -> tuple[np.ndarray, np.ndarray]:
    """Compute yield intercept and factor loadings."""
    n_maturities = len(tau)
    b_mat = np.zeros((n_maturities, 3), dtype=np.float64)
    a_vec = np.zeros(n_maturities, dtype=np.float64)

    for m in range(n_maturities):
        tm = float(tau[m])
        for i in range(3):
            ki = float(kappa[i])
            si = float(sigma[i])
            b_mat[m, i] = -(1.0 - np.exp(-ki * tm)) / ki
            int_b2 = (1.0 / (ki**2)) * (
                tm - 2.0 * (1.0 - np.exp(-ki * tm)) / ki + (1.0 - np.exp(-2.0 * ki * tm)) / (2.0 * ki)
            )
            a_vec[m] += 0.5 * (si**2) * int_b2
        a_vec[m] -= delta0 * tm

    a_yield = -a_vec / tau
    h_yield = -b_mat / tau[:, np.newaxis]
    return a_yield, h_yield


def run_kalman_filter(
    yields: np.ndarray,
    tau: np.ndarray,
    delta0: float,
    kappa: np.ndarray,
    sigma: np.ndarray,
    c_vec: np.ndarray,
    phi_mat: np.ndarray,
    q_cov: np.ndarray,
    meas_std: float,
) -> tuple[float, float, np.ndarray]:
    """Run Kalman filter recursion and return log-likelihood, pricing RMSE, and filtered states."""
    n_periods, n_maturities = yields.shape
    n_states = 3

    a_yield, h_yield = solve_loadings(tau, delta0, kappa, sigma)
    r_cov = np.eye(n_maturities) * (meas_std**2)

    # Stationary covariance: V0 = Phi V0 Phi^T + Q
    i_kron = np.eye(n_states * n_states) - np.kron(phi_mat, phi_mat)
    cov_x0 = np.linalg.solve(i_kron, q_cov.flatten()).reshape((n_states, n_states))
    cov_x0 = 0.5 * (cov_x0 + cov_x0.T)
    mean_x0 = np.linalg.solve(np.eye(n_states) - phi_mat, c_vec)

    x_curr = mean_x0.copy()
    p_curr = cov_x0.copy()

    log_likelihood = 0.0
    filtered_states = np.zeros((n_periods, n_states), dtype=np.float64)
    pricing_errors = np.zeros((n_periods, n_maturities), dtype=np.float64)
    two_pi_m = n_maturities * np.log(2.0 * np.pi)

    for t in range(n_periods):
        # 1. Predict
        x_pred = c_vec + phi_mat @ x_curr
        p_pred = phi_mat @ p_curr @ phi_mat.T + q_cov

        # 2. Measurement
        y_pred = a_yield + h_yield @ x_pred
        v_t = yields[t] - y_pred

        # 3. Covariance & likelihood
        f_cov = h_yield @ p_pred @ h_yield.T + r_cov
        f_inv_v = np.linalg.solve(f_cov, v_t)
        _sign, logdet = np.linalg.slogdet(f_cov)
        log_likelihood += -0.5 * (two_pi_m + logdet + float(v_t @ f_inv_v))

        # 4. Update
        k_gain = (np.linalg.solve(f_cov, h_yield @ p_pred.T)).T
        x_curr = x_pred + k_gain @ v_t
        p_curr = (np.eye(n_states) - k_gain @ h_yield) @ p_pred
        p_curr = 0.5 * (p_curr + p_curr.T)

        filtered_states[t] = x_curr
        pricing_errors[t] = yields[t] - (a_yield + h_yield @ x_curr)

    rmse_bps = float(np.sqrt(np.mean(pricing_errors**2)) * 10_000.0)
    return float(log_likelihood), rmse_bps, filtered_states


def main() -> None:
    rng = np.random.default_rng(101)
    n_periods = 60
    tau = np.array([0.25, 0.5, 1.0, 2.0, 5.0, 10.0], dtype=np.float64)
    n_maturities = len(tau)

    # True ATSM parameters
    delta0 = 0.024
    kappa = np.array([0.055, 0.26, 0.82], dtype=np.float64)
    sigma = np.array([0.011, 0.015, 0.019], dtype=np.float64)

    c_vec = np.zeros(3, dtype=np.float64)
    phi_mat = np.diag([0.98, 0.92, 0.80])
    dt = 1.0 / 12.0
    q_cov = np.diag(sigma**2 * dt)
    meas_std = 0.0005  # 5 bps

    a_yield, h_yield = solve_loadings(tau, delta0, kappa, sigma)

    # Generate synthetic observations
    x_sim = np.zeros((n_periods, 3), dtype=np.float64)
    yields = np.zeros((n_periods, n_maturities), dtype=np.float64)
    for t in range(1, n_periods):
        x_sim[t] = phi_mat @ x_sim[t - 1] + rng.multivariate_normal(np.zeros(3), q_cov)
    for t in range(n_periods):
        yields[t] = a_yield + h_yield @ x_sim[t] + rng.normal(0.0, meas_std, size=n_maturities)

    ll, rmse_bps, x_filt = run_kalman_filter(yields, tau, delta0, kappa, sigma, c_vec, phi_mat, q_cov, meas_std)

    print("SIMULATED 3-Factor ATSM Kalman Pipeline (Worked Example Rung 1):")
    print(f"  Observed yield curve: {n_periods} periods x {n_maturities} tenors")
    print(f"  Short rate baseline delta0: {delta0:.4f}")
    print(f"  Mean reversion speeds kappa: {[round(float(k), 3) for k in kappa]}")
    print(f"  Volatilities sigma: {[round(float(s), 4) for s in sigma]}")
    print(f"  Log-likelihood: {ll:.2f}")
    print(f"  Pricing error RMSE: {rmse_bps:.2f} bps")
    print("Factor | True Mean | Filtered Mean | Correlation")
    print("-" * 50)
    for i in range(3):
        corr = float(np.corrcoef(x_filt[:, i], x_sim[:, i])[0, 1])
        s_m = float(np.mean(x_sim[:, i]))
        f_m = float(np.mean(x_filt[:, i]))
        print(f"   {i + 1}   |  {s_m:+.4f}   |    {f_m:+.4f}     |   {corr:.4f}")


main()

Specification

Context

You are a senior quantitative rates researcher on a fixed-income systematic trading desk. Your mandate is to calibrate a canonical three-factor Gaussian Affine Term-Structure Model (Dai and Singleton 2000, Section 2) to historical Treasury yield curves. You must estimate latent factor trajectories (level, slope, curvature) via Kalman-filter maximum likelihood estimation (Hamilton 1994, Chapter 13), decompose cross-sectional pricing errors across maturities, and author an institutional model validation memo assessing parameter identification, innovation whiteness, and model risk under stress.

Inputs

The autograder supplies an instance dictionary generated deterministically from an integer seed via make_instance(seed: int) in instance.py. The dictionary contains:

  • T: int, number of monthly observation periods (e.g. 60).
  • M: int, number of yield maturities (e.g. 6).
  • tau: list of float, maturities in years: [0.25, 0.5, 1.0, 2.0, 5.0, 10.0].
  • delta0: float, baseline instantaneous short rate intercept.
  • kappa: list of float, true risk-neutral mean reversion speeds \([\kappa_1, \kappa_2, \kappa_3]\).
  • sigma: list of float, true factor volatilities \([\sigma_1, \sigma_2, \sigma_3]\).
  • c: list of float, physical state intercept vector.
  • phi: list of list of float, physical state transition matrix \(\boldsymbol{\Phi}\).
  • Q: list of list of float, state shock covariance matrix \(\mathbf{Q}\).
  • R_diag: list of float, measurement error variances for observed yields.
  • true_states: list of list of float, true simulated latent factor path of shape \((T, 3)\).
  • yields: list of list of float, observed synthetic yield curve matrix of shape \((T, M)\).
Required outputs

A module file affine_calibration.py exporting the following functions with strict type annotations:

  • solve_riccati_ode(tau, delta0, kappa, sigma): returns (A, B) with shapes \((M,)\) and \((M, 3)\).
  • yield_loadings(tau, delta0, kappa, sigma): returns (a, H) with shapes \((M,)\) and \((M, 3)\).
  • compute_unconditional_moments(c, phi, q_cov): solves the discrete Lyapunov equation returning unconditional mean and covariance (mean_X, cov_X).
  • kalman_filter(yields, tau, delta0, kappa, sigma, c, phi, q_cov, r_diag): executes the full Kalman filter recursion returning a dictionary with keys log_likelihood, filtered_states, predicted_yields, innovations, pricing_errors, and rmse_bps.
  • kalman_log_likelihood(params, yields, tau, c, phi, q_cov, r_diag): evaluates the scalar log-likelihood for numerical optimization.
  • calibrate_mle(yields, tau, initial_params=None): performs bounded optimization to recover calibrated parameters and pricing diagnostics.

An institutional model validation memo memo.md (approximately 1,000–1,500 words) diagnosing parameter identification, innovation orthogonality, and pricing error structures.

Constraints

Standard library, NumPy, and SciPy only. All matrix operations must use numerically stable solvers (e.g. np.linalg.solve). Code must be clean under ruff check, ruff format --check, and mypy --strict without warnings. The calibration must finish within 5 seconds per seed.

    a_true, h_true = solve_riccati(tau, true_delta0, true_kappa, true_sigma)

    x_true = np.zeros((n_periods, 3), dtype=np.float64)
    yields = np.zeros((n_periods, n_maturities), dtype=np.float64)
    for t in range(1, n_periods):
        x_true[t] = phi_mat @ x_true[t - 1] + rng.multivariate_normal(np.zeros(3), q_cov)
    for t in range(n_periods):
        yields[t] = a_true + h_true @ x_true[t] + rng.normal(0.0, meas_std, size=n_maturities)

    # Initial guess for optimization
    p0 = np.concatenate([[true_delta0 * 1.05], true_kappa * 0.95, true_sigma])

    def objective(p: np.ndarray) -> float:
        d0 = float(p[0])
        kp = p[1:4]
        sg = p[4:7]
        if np.any(kp <= 0) or np.any(sg <= 0):
            return 1e12
        try:
            ll, _, _, _ = run_kalman_filter(yields, tau, d0, kp, sg, c_vec, phi_mat, q_cov, r_cov)
            return -ll
        except Exception:
            return 1e12

    bounds = [
        (0.01, 0.10),
        (0.01, 0.50),
        (0.10, 1.50),
        (0.40, 3.00),
        (0.005, 0.05),
        (0.005, 0.05),
        (0.005, 0.05),
    ]

    opt = optimize.minimize(objective, p0, method="L-BFGS-B", bounds=bounds, options={"maxiter": 20})
    p_opt = opt.x
    delta0_cal = float(p_opt[0])
    kappa_cal = p_opt[1:4]
    sigma_cal = p_opt[4:7]

    ll_cal, rmse_cal, _x_filt, errors = run_kalman_filter(
        yields, tau, delta0_cal, kappa_cal, sigma_cal, c_vec, phi_mat, q_cov, r_cov
    )

    print("SIMULATED 3-Factor ATSM Calibration Validation (Rung 3):")

    print(f"  Calibrated delta0 = {delta0_cal:.4f} (True: {true_delta0:.4f})")
    print(f"  Calibrated kappa  = {[round(float(k), 3) for k in kappa_cal]} (True: {true_kappa.tolist()})")
    print(f"  Overall Yield Pricing RMSE: {rmse_cal:.2f} bps")

2. Mathematical Structure and Identification: The instantaneous short rate is parameterized as \(r_t = \delta_0 + \boldsymbol{\delta}_1^\top X_t\) with \(\boldsymbol{\delta}_1 = [1, 1, 1]^\top\). Under \(\mathbb{Q}\), the factor diffusion follows \(dX_t = -K^\mathbb{Q} X_t dt + \Sigma dW_t^\mathbb{Q}\), with \(K^\mathbb{Q} = \operatorname{diag}(\kappa_1, \kappa_2, \kappa_3)\) and \(\Sigma = \operatorname{diag}(\sigma_1, \sigma_2, \sigma_3)\). Zero-coupon bond prices satisfy \(P(t, \tau) = \exp(A(\tau) + B(\tau)^\top X_t)\), where the Riccati ODE system (Duffie and Kan 1996, Theorem 1) admits closed-form analytical solutions: \[ B_i(\tau) = -\frac{1 - e^{-\kappa_i \tau}}{\kappa_i}, \quad A(\tau) = -\delta_0 \tau + \frac{1}{2} \sum_{i=1}^3 \frac{\sigma_i^2}{\kappa_i^2}\left[\tau - \frac{2(1 - e^{-\kappa_i \tau})}{\kappa_i} + \frac{1 - e^{-2\kappa_i \tau}}{2\kappa_i}\right] \] Yield factor loadings \(\mathbf{H}_{m, i} = -B_i(\tau_m)/\tau_m\) decay strictly monotonically with maturity. Factor 1 exhibits a half-life of \((\ln 2)/\kappa_1 = 11.6\) years, acting as a level shift; Factor 2 has a half-life of 2.5 years (slope); and Factor 3 has a half-life of 0.8 years (curvature).
