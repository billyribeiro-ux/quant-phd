Meese–Rogoff Re-Run

In their landmark empirical study, Meese & Rogoff (1983) demonstrated that structural macroeconomic models of exchange rates fail to out-predict a driftless random walk in out-of-sample forecasting, even when the structural models are endowed with actual realized future values of explanatory fundamentals. This project constructs an empirical forecasting engine (meese_rogoff.py) that replicates the Meese–Rogoff horse race across multiple forecast horizons (1, 3, and 12 months) under strict rolling-window real-time discipline. You will compute Root Mean Squared Errors, Theil's U inequality coefficients, and Diebold–Mariano predictive accuracy statistics, and author an analytical research memo (report.md) evaluated by rubric.

"""Lab X1-project-1: Rung 1 worked example for Meese-Rogoff re-run.

Replays the out-of-sample forecasting horse race comparing a driftless random walk
against macroeconomic fundamentals and UIP models across 1, 3, and 12-month horizons.
"""

from __future__ import annotations

import math
import random
from typing import Any


def make_instance(seed: int) -> dict[str, Any]:
    """Generate a deterministic problem instance for the X1 project."""
    rng = random.Random(seed)
    n_obs = 120 + (seed * 11) % 41
    initial_price = round(1.10 + ((seed * 7) % 30) * 0.01, 4)
    log_s0 = math.log(initial_price)
    window_size = 36
    horizons = [1, 3, 12]

    rho_f = round(0.92 + (seed % 6) * 0.01, 3)
    sigma_f = 0.02
    f_series: list[float] = [round(rng.gauss(0.0, 0.05), 4)]
    for _ in range(1, n_obs):
        f_next = rho_f * f_series[-1] + rng.gauss(0.0, sigma_f)
        f_series.append(round(f_next, 4))

    i_diff_series: list[float] = []
    for t in range(n_obs):
        diff = round(0.0015 * f_series[t] + rng.gauss(0.001, 0.0008), 5)
        i_diff_series.append(diff)

    beta_true = round(0.12 + (seed % 5) * 0.03, 3)
    sigma_s = 0.03
    s_series: list[float] = [round(log_s0, 4)]
    for t in range(1, n_obs):
        shock = rng.gauss(0.0, sigma_s)
        f_delta = f_series[t] - f_series[t - 1]
        ds = beta_true * f_delta + shock
        s_next = s_series[-1] + ds
        s_series.append(round(s_next, 4))

    return {
        "seed": seed,
        "n_obs": n_obs,
        "initial_price": initial_price,
        "window_size": window_size,
        "horizons": horizons,
        "s_series": s_series,
        "f_series": f_series,
        "i_diff_series": i_diff_series,
    }


def rolling_forecast_rw(s_series: list[float], horizon: int, window_size: int) -> list[float]:
    """Compute out-of-sample driftless random walk forecasts E_t[s_{t+h}] = s_t."""
    n = len(s_series)
    return [round(s_series[t], 6) for t in range(window_size, n - horizon)]


def rolling_forecast_fundamentals(
    s_series: list[float],
    f_series: list[float],
    horizon: int,
    window_size: int,
) -> list[float]:
    """Compute rolling OLS monetary fundamentals forecasts using realized fundamentals."""
    n = len(s_series)
    forecasts: list[float] = []
    for t in range(window_size, n - horizon):
        s_win = s_series[t - window_size : t]
        f_win = f_series[t - window_size : t]
        mean_s = sum(s_win) / window_size
        mean_f = sum(f_win) / window_size
        var_f = sum((f - mean_f) ** 2 for f in f_win)
        cov_sf = sum((f - mean_f) * (s - mean_s) for s, f in zip(s_win, f_win, strict=True))
        beta_hat = cov_sf / var_f if var_f > 1e-12 else 0.0
        alpha_hat = mean_s - beta_hat * mean_f
        pred = alpha_hat + beta_hat * f_series[t + horizon]
        forecasts.append(round(pred, 6))
    return forecasts


def rolling_forecast_uip(
    s_series: list[float],
    i_diff_series: list[float],
    horizon: int,
    window_size: int,
) -> list[float]:
    """Compute out-of-sample UIP forecasts E_t[s_{t+h}] = s_t + h * (i_t - i*_t)."""
    n = len(s_series)
    return [round(s_series[t] + float(horizon) * i_diff_series[t], 6) for t in range(window_size, n - horizon)]


def compute_metrics(actual: list[float], predicted: list[float]) -> dict[str, float]:
    """Compute RMSE and MAE."""
    m = len(actual)
    sse = sum((a - p) ** 2 for a, p in zip(actual, predicted, strict=True))
    sae = sum(abs(a - p) for a, p in zip(actual, predicted, strict=True))
    return {"rmse": round(math.sqrt(sse / m), 6), "mae": round(sae / m, 6)}


def compute_diebold_mariano(
    actual: list[float],
    pred_model: list[float],
    pred_bench: list[float],
    horizon: int,
) -> float:
    """Compute Diebold-Mariano test statistic with Bartlett HAC adjustment."""
    t_eval = len(actual)
    d = [(a - pm) ** 2 - (a - pb) ** 2 for a, pm, pb in zip(actual, pred_model, pred_bench, strict=True)]
    mean_d = sum(d) / t_eval
    gamma: list[float] = []
    max_lag = max(1, horizon - 1)
    for k in range(max_lag + 1):
        if k == 0:
            g = sum((val - mean_d) ** 2 for val in d) / t_eval
        else:
            g = sum((d[t] - mean_d) * (d[t - k] - mean_d) for t in range(k, t_eval)) / t_eval
        gamma.append(g)
    lr_var = gamma[0]
    if horizon > 1:
        for k in range(1, horizon):
            weight = 1.0 - (k / float(horizon))
            lr_var += 2.0 * weight * gamma[k]
    variance_dbar = lr_var / t_eval
    if variance_dbar <= 1e-12:
        return 0.0
    return round(mean_d / math.sqrt(variance_dbar), 4)


def run_meese_rogoff(instance: dict[str, Any]) -> dict[str, Any]:
    """Execute the complete Meese-Rogoff horse race across all horizons."""
    s_series = [float(x) for x in instance["s_series"]]
    f_series = [float(x) for x in instance["f_series"]]
    i_diff_series = [float(x) for x in instance["i_diff_series"]]
    window_size = int(instance["window_size"])
    horizons = [int(h) for h in instance["horizons"]]
    n_obs = len(s_series)
    results_by_horizon: dict[int, dict[str, Any]] = {}

    for h in horizons:
        actual = [s_series[t + h] for t in range(window_size, n_obs - h)]
        rw_preds = rolling_forecast_rw(s_series, h, window_size)
        f_preds = rolling_forecast_fundamentals(s_series, f_series, h, window_size)
        uip_preds = rolling_forecast_uip(s_series, i_diff_series, h, window_size)

        rw_metrics = compute_metrics(actual, rw_preds)
        f_metrics = compute_metrics(actual, f_preds)
        uip_metrics = compute_metrics(actual, uip_preds)

        u_fund = round(f_metrics["rmse"] / rw_metrics["rmse"], 4)
        u_uip = round(uip_metrics["rmse"] / rw_metrics["rmse"], 4)
        dm_fund = compute_diebold_mariano(actual, f_preds, rw_preds, h)
        dm_uip = compute_diebold_mariano(actual, uip_preds, rw_preds, h)

        results_by_horizon[h] = {
            "eval_count": len(actual),
            "rw_rmse": rw_metrics["rmse"],
            "rw_mae": rw_metrics["mae"],
            "fund_rmse": f_metrics["rmse"],
            "fund_mae": f_metrics["mae"],
            "fund_theil_u": u_fund,
            "fund_dm_stat": dm_fund,
            "uip_rmse": uip_metrics["rmse"],
            "uip_mae": uip_metrics["mae"],
            "uip_theil_u": u_uip,
            "uip_dm_stat": dm_uip,
        }

    return {
        "seed": int(instance["seed"]),
        "n_obs": n_obs,
        "window_size": window_size,
        "horizons": horizons,
        "results": results_by_horizon,
    }


def run_worked_example() -> None:
    print("Step 1 | Generating problem instance (seed 42, N=130 observations)")
    inst = make_instance(42)
    s_series = inst["s_series"]
    f_series = inst["f_series"]

    )

            f"{hr['fund_theil_u']:>14.4f} | {hr['fund_dm_stat']:>12.2f} | {hr['uip_theil_u']:>13.4f}"

The theoretical model and horse-race framework

The forecasting framework formalizes the structural exchange-rate horse race established by Meese & Rogoff (1983) and reviewed by Obstfeld & Rogoff (1996).

Assumptions

  1. A1 · Log exchange rate decomposition. The spot exchange rate \(S_t\) in log terms is \(s_t = \ln S_t\). The true data-generating process satisfies \(s_t = s_{t-1} + \beta \Delta f_t + \epsilon_t\), where \(f_t\) is an observable macroeconomic fundamental and \(\epsilon_t \sim \text{i.i.d.} \mathcal{N}(0, \sigma^2)\).
  2. A2 · Driftless random-walk benchmark. Under the efficient markets hypothesis and martingales, the optimal \(h\)-step-ahead point forecast is the current spot rate: \(\hat{s}_{t+h|t}^{\text{RW}} = s_t\).
  3. A3 · Structural fundamentals rolling regression. On each rolling estimation window \([t - W, t)\) of length \(W \ge 10\), an ordinary least squares regression estimates \(s_\tau = \alpha + \beta f_\tau + u_\tau\). Following Meese & Rogoff (1983), the out-of-sample forecast is evaluated using realized fundamentals: \(\hat{s}_{t+h|t}^{\text{Fund}} = \hat{\alpha}_t + \hat{\beta}_t f_{t+h}\).
  4. A4 · Uncovered interest parity forward-rate benchmark. Under UIP, the expected exchange rate change equals the interest rate differential: \(\hat{s}_{t+h|t}^{\text{UIP}} = s_t + h \cdot (i_t - i_t^*)\).
  5. A5 · Quadratic loss and Diebold–Mariano inference. Forecast accuracy is evaluated under squared error loss \(L(e) = e^2\). For horizon \(h \ge 1\), the loss differential \(d_t = (e_{\text{model}, t})^2 - (e_{\text{RW}, t})^2\) is tested using the Diebold–Mariano statistic with a Bartlett kernel weighting up to lag \(h - 1\).

Theoretical propositions

Proposition 1 · In-sample fit vs out-of-sample predictive failure. Under Assumptions A1–A3, even if the structural model achieves high in-sample explanatory power (\(R^2 > 0.5\)), sampling variation in \(\hat{\beta}_t\) under a rolling window of length \(W\) inflates out-of-sample forecast variance by \(\Var(\hat{\beta}_t) \cdot \Var(f_{t+h})\), causing the structural model's out-of-sample RMSE to exceed that of the driftless random walk (Theil's \(U > 1.0\)).

Proposition 2 · Asymptotic horizon convergence. Under Assumptions A1–A5, as the forecast horizon \(h\) expands toward long horizons (\(h \to \infty\)), the cumulative variance of random-walk innovations \(\Var(s_{t+h} - s_t) = h \sigma^2\) grows linearly, whereas persistent cointegrating fundamental mean-reversion stabilizes \(\Var(s_{t+h} - \alpha - \beta f_{t+h})\), leading Theil's U to converge toward or below unity at multi-year horizons.

    sigma_f = 0.02
    f_series: list[float] = [round(rng.gauss(0.0, 0.05), 4)]
    for _ in range(1, n_obs):
        f_next = rho_f * f_series[-1] + rng.gauss(0.0, sigma_f)
        f_series.append(round(f_next, 4))

    i_diff_series: list[float] = []
    for t in range(n_obs):
        diff = round(0.0015 * f_series[t] + rng.gauss(0.001, 0.0008), 5)
        i_diff_series.append(diff)

    beta_true = round(0.12 + (seed % 5) * 0.03, 3)
    sigma_s = 0.03
    s_series: list[float] = [round(log_s0, 4)]
    for t in range(1, n_obs):
        shock = rng.gauss(0.0, sigma_s)
        f_delta = f_series[t] - f_series[t - 1]
        ds = beta_true * f_delta + shock
        s_next = s_series[-1] + ds
        s_series.append(round(s_next, 4))

    return {
        "seed": seed,
        "n_obs": n_obs,
        "initial_price": initial_price,
        "window_size": window_size,
        "horizons": horizons,
        "s_series": s_series,
        "f_series": f_series,
        "i_diff_series": i_diff_series,
    }


def rolling_forecast_rw(s_series: list[float], horizon: int, window_size: int) -> list[float]:
    """Compute out-of-sample driftless random walk forecasts E_t[s_{t+h}] = s_t."""
    n = len(s_series)
    return [round(s_series[t], 6) for t in range(window_size, n - horizon)]


def rolling_forecast_fundamentals(
    s_series: list[float],
    f_series: list[float],
    horizon: int,
    window_size: int,
) -> list[float]:
    """Compute rolling OLS monetary fundamentals forecasts using realized fundamentals."""
    n = len(s_series)
    forecasts: list[float] = []
    for t in range(window_size, n - horizon):
        s_win = s_series[t - window_size : t]
        f_win = f_series[t - window_size : t]
        mean_s = sum(s_win) / window_size
        mean_f = sum(f_win) / window_size
        var_f = sum((f - mean_f) ** 2 for f in f_win)
        cov_sf = sum((f - mean_f) * (s - mean_s) for s, f in zip(s_win, f_win, strict=True))
        beta_hat = cov_sf / var_f if var_f > 1e-12 else 0.0
        alpha_hat = mean_s - beta_hat * mean_f
        pred = alpha_hat + beta_hat * f_series[t + horizon]
        forecasts.append(round(pred, 6))
    return forecasts


def rolling_forecast_uip(
    s_series: list[float],
    i_diff_series: list[float],
    horizon: int,
    window_size: int,
) -> list[float]:
    """Compute out-of-sample UIP forecasts E_t[s_{t+h}] = s_t + h * (i_t - i*_t)."""
    n = len(s_series)
    return [round(s_series[t] + float(horizon) * i_diff_series[t], 6) for t in range(window_size, n - horizon)]


def compute_metrics(actual: list[float], predicted: list[float]) -> dict[str, float]:
    """Compute RMSE and MAE."""
    m = len(actual)
    sse = sum((a - p) ** 2 for a, p in zip(actual, predicted, strict=True))
    sae = sum(abs(a - p) for a, p in zip(actual, predicted, strict=True))
    return {"rmse": round(math.sqrt(sse / m), 6), "mae": round(sae / m, 6)}


def compute_diebold_mariano(
    actual: list[float],
    pred_model: list[float],
    pred_bench: list[float],
    horizon: int,
) -> float:
    """Compute Diebold-Mariano test statistic with Bartlett HAC adjustment."""
    t_eval = len(actual)
    d = [(a - pm) ** 2 - (a - pb) ** 2 for a, pm, pb in zip(actual, pred_model, pred_bench, strict=True)]
    mean_d = sum(d) / t_eval
    gamma: list[float] = []
    max_lag = max(1, horizon - 1)
    for k in range(max_lag + 1):
        if k == 0:
            g = sum((val - mean_d) ** 2 for val in d) / t_eval
        else:
            g = sum((d[t] - mean_d) * (d[t - k] - mean_d) for t in range(k, t_eval)) / t_eval
        gamma.append(g)
    lr_var = gamma[0]
    if horizon > 1:
        for k in range(1, horizon):
            weight = 1.0 - (k / float(horizon))
            lr_var += 2.0 * weight * gamma[k]
    variance_dbar = lr_var / t_eval
    if variance_dbar <= 1e-12:
        return 0.0
    return round(mean_d / math.sqrt(variance_dbar), 4)


def run_meese_rogoff(instance: dict[str, Any]) -> dict[str, Any]:
    """Execute the complete Meese-Rogoff horse race across all horizons."""
    s_series = [float(x) for x in instance["s_series"]]
    f_series = [float(x) for x in instance["f_series"]]
    i_diff_series = [float(x) for x in instance["i_diff_series"]]
    window_size = int(instance["window_size"])
    horizons = [int(h) for h in instance["horizons"]]
    n_obs = len(s_series)
    results_by_horizon: dict[int, dict[str, Any]] = {}

    for h in horizons:
        actual = [s_series[t + h] for t in range(window_size, n_obs - h)]
        rw_preds = rolling_forecast_rw(s_series, h, window_size)
        f_preds = rolling_forecast_fundamentals(s_series, f_series, h, window_size)
        uip_preds = rolling_forecast_uip(s_series, i_diff_series, h, window_size)

        rw_metrics = compute_metrics(actual, rw_preds)
        f_metrics = compute_metrics(actual, f_preds)
        uip_metrics = compute_metrics(actual, uip_preds)

        u_fund = round(f_metrics["rmse"] / rw_metrics["rmse"], 4)
        u_uip = round(uip_metrics["rmse"] / rw_metrics["rmse"], 4)
        dm_fund = compute_diebold_mariano(actual, f_preds, rw_preds, h)
        dm_uip = compute_diebold_mariano(actual, uip_preds, rw_preds, h)

        results_by_horizon[h] = {
            "eval_count": len(actual),
            "rw_rmse": rw_metrics["rmse"],
            "rw_mae": rw_metrics["mae"],
            "fund_rmse": f_metrics["rmse"],
            "fund_mae": f_metrics["mae"],
            "fund_theil_u": u_fund,
            "fund_dm_stat": dm_fund,
            "uip_rmse": uip_metrics["rmse"],
            "uip_mae": uip_metrics["mae"],
            "uip_theil_u": u_uip,
            "uip_dm_stat": dm_uip,
        }

    return {
        "seed": int(instance["seed"]),
        "n_obs": n_obs,
        "window_size": window_size,
        "horizons": horizons,
        "results": results_by_horizon,
    }

Model specification

We evaluate bilateral exchange rate predictability following the empirical architecture of Meese & Rogoff (1983). The economy satisfies Assumptions A1–A5: log exchange rates \(s_t = \ln S_t\) follow a persistent process with fundamental loading \(\beta\) and noise \(\sigma = 0.03\). On each rolling window \([t - W, t)\) with \(W = 36\) months, we estimate OLS regressions to forecast \(s_{t+h}\) across horizons \(h \in \{1, 3, 12\}\). We compare the driftless random walk \(\hat{s}_{t+h|t}^{\text{RW}} = s_t\) against the structural fundamentals model \(\hat{s}_{t+h|t}^{\text{Fund}} = \hat{\alpha}_t + \hat{\beta}_t f_{t+h}\) and the uncovered interest parity benchmark \(\hat{s}_{t+h|t}^{\text{UIP}} = s_t + h \cdot (i_t - i_t^*)\).

Estimators and identities

Out-of-sample forecast accuracy is quantified via Root Mean Squared Error \(\text{RMSE} = \sqrt{\frac{1}{M}\sum_{t}(s_{t+h} - \hat{s}_{t+h|t})^2}\) and Theil's inequality coefficient \(U = \text{RMSE}_{\text{model}} / \text{RMSE}_{\text{RW}}\) (Proposition 1). Forecast superiority is tested using the Diebold–Mariano statistic \(DM = \bar{d} / \sqrt{\hat{V}(\bar{d})}\), where \(d_t = (e_{\text{model}, t})^2 - (e_{\text{RW}, t})^2\) and \(\hat{V}(\bar{d})\) applies Bartlett triangular weights \((1 - k/h)\) for multi-step horizons \(h > 1\). The solver validates seven boundary conditions, raising ValueError on non-positive prices, non-positive horizons, windows \(W < 10\), insufficient observations (\(N \le W + h\)), or mismatched series lengths.

Results and horse-race analysis

The Lab 3 benchmark grid confirms the Meese–Rogoff puzzle across both seeds. On Seed 0, at the 1-month horizon (\(M = 83\)), the random walk achieves \(\text{RMSE} = 0.0301\) compared to \(0.0687\) for fundamentals, yielding Theil's \(U = 2.2832\). The Diebold–Mariano statistic of \(4.57\) decisively rejects equal predictive accuracy in favor of the random walk (\(p < 0.001\)). On Seed 1 (\(M = 94\)), the 1-month Theil's U is even higher at \(2.5824\) (\(DM = 6.92\)). Across both seeds, UIP performs closely to the random walk (Theil's \(U \approx 0.997\) to \(1.007\)). Crucially, as the horizon expands to 12 months, Theil's U for the fundamentals model declines from \(2.2832\) to \(1.0311\) on Seed 0 (DM statistic drops from \(4.57\) to \(0.44\)), confirming Proposition 2: persistent fundamental mean-reversion reasserts itself at long horizons as transitory noise averages out.

Numerical rigour and rounding

All internal computations are executed in IEEE 754 64-bit floating-point precision. Output values are reported to four decimal places. As a manual verification check, consider the first evaluation point at \(t = 36\) for Seed 0 (\(h = 1\)): the actual target is \(s_{37} = 0.0731\), while the random walk forecast is \(s_{36} = 0.0479\), yielding squared error \((0.0731 - 0.0479)^2 = 0.000635\). Repeating across all 83 evaluation points yields cumulative sum of squared errors \(0.07521\), giving \(\text{RMSE} = \sqrt{0.07521 / 83} = 0.030102 \approx 0.0301\), matching the solver output exactly.

Limits and evidence

While the random walk benchmark remains exceptionally robust (Meese & Rogoff 1983), three structural limitations must be emphasized: (1) Ex-post vs real-time data: Meese & Rogoff used ex-post revised macroeconomic data, giving structural models an unfair advantage; real-time unrevised vintages degrade performance further; (2) Transaction costs and non-linearities: Real exchange rate adjustment exhibits threshold behavior due to shipping costs and trade barriers, so linear OLS understates mean-reversion speed (Obstfeld & Rogoff 1996); (3) Time-varying risk premia: UIP failure reflects volatile risk premia that standard monetary models omit (Fama 1984; Du et al. 2018).
