Currency Factor Replication

In foundational empirical asset pricing, Lustig, Roussanov & Verdelhan (2011) proved that cross-sectional currency excess returns are governed by two risk factors: a level factor (the Dollar factor, \(R_{\mathrm{DOL}}\)) capturing broad dollar fluctuations, and a slope factor (High-Minus-Low FX carry, \(\mathrm{HML}_{\mathrm{FX}}\)) pricing the cross-section of interest rate-sorted portfolios. Subsequently, Menkhoff, Sarno, Schmeling & Schrimpf (2012) demonstrated that innovations in global foreign exchange volatility (\(\Delta \mathrm{Vol}_{\mathrm{FX}}\)) price the carry cross-section because high-carry currencies load negatively on volatility risk. In this project, you will build an end-to-end econometric factor replication engine (currency_factors.py) and author an analytical research memo (report.md) evaluated by rubric.

Rung 1 · Worked example

In this worked example, a cross-sectional panel of 10 currencies over 120 months (seed 42) is sorted into 5 carry quintile portfolios based on lagged forward discounts. We construct the Dollar factor and the HML-FX factor, estimate portfolio factor betas, and run two-pass Fama–MacBeth regressions.

Lab 1 · Currency factor replication worked example

Sorting currencies into quintiles will produce a monotonic increase in annualized excess returns from Portfolio 1 to Portfolio 5, generating a positive HML-FX carry spread. In time-series regressions, Dollar factor betas will cluster near 1.0 across all portfolios, while HML-FX betas will increase monotonically from negative for Portfolio 1 to positive for Portfolio 5, yielding a cross-sectional \(R^2 > 0.85\).

"""Lab X2-project-1: Rung 1 Worked Example for Currency Factor Replication.

Traces complete replication of Lustig et al. (2011) Dollar and HML-FX factors
and Menkhoff et al. (2012) global FX volatility innovations on seed 42.
"""

from __future__ import annotations

import math
import random
from typing import Any


def _solve_linear_system(a_matrix: list[list[float]], b_vector: list[float]) -> list[float]:
    n = len(b_vector)
    aug = [[*row[:], b_vector[i]] for i, row in enumerate(a_matrix)]
    for col in range(n):
        max_row = col
        max_val = abs(aug[col][col])
        for row in range(col + 1, n):
            if abs(aug[row][col]) > max_val:
                max_val = abs(aug[row][col])
                max_row = row
        if max_val < 1e-12:
            raise ValueError("Matrix is singular")
        aug[col], aug[max_row] = aug[max_row], aug[col]
        pivot = aug[col][col]
        for c in range(col, n + 1):
            aug[col][c] /= pivot
        for row in range(n):
            if row != col:
                factor = aug[row][col]
                for c in range(col, n + 1):
                    aug[row][c] -= factor * aug[col][c]
    return [aug[i][n] for i in range(n)]


def _multivariate_ols(
    x_matrix: list[list[float]],
    y_vector: list[float],
    fit_intercept: bool = True,
) -> tuple[list[float], list[float], float]:
    t_obs = len(y_vector)
    p_vars = len(x_matrix[0]) if x_matrix else 0
    if fit_intercept:
        design = [[1.0, *row] for row in x_matrix]
        num_params = p_vars + 1
    else:
        design = [row[:] for row in x_matrix]
        num_params = p_vars

    xt_x = [[0.0] * num_params for _ in range(num_params)]
    xt_y = [0.0] * num_params
    for t in range(t_obs):
        row = design[t]
        y_val = y_vector[t]
        for i in range(num_params):
            xt_y[i] += row[i] * y_val
            for j in range(num_params):
                xt_x[i][j] += row[i] * row[j]

    coeffs = _solve_linear_system(xt_x, xt_y)
    fitted: list[float] = []
    for t in range(t_obs):
        pred = sum(design[t][i] * coeffs[i] for i in range(num_params))
        fitted.append(pred)

    y_mean = sum(y_vector) / t_obs
    ss_tot = sum((y - y_mean) ** 2 for y in y_vector)
    ss_res = sum((y - fitted[t]) ** 2 for t, y in enumerate(y_vector))
    r_squared = 1.0 - (ss_res / ss_tot) if ss_tot > 1e-12 else 0.0
    return coeffs, fitted, r_squared


def run_worked_example(seed: int = 42) -> dict[str, Any]:
    """Execute complete currency factor replication on seed 42."""
    rng = random.Random(seed)
    n_periods = 120
    n_cur = 10
    n_portfolios = 5

    base_spreads = [0.025, 0.005, -0.020, -0.005, 0.010, -0.035, 0.015, 0.030, 0.008, 0.060]
    initial_spots = [0.70, 0.75, 1.10, 1.15, 1.30, 0.009, 0.11, 0.65, 0.10, 0.065]
    log_spots = [math.log(s) for s in initial_spots]
    days_per_month = 21

    spot_rates: list[list[float]] = []
    forward_rates: list[list[float]] = []
    daily_returns: list[list[float]] = []
    vol_state = 0.006

    for _ in range(n_periods):
        cur_spots = [round(math.exp(ls), 5) for ls in log_spots]
        spot_rates.append(cur_spots)
        month_forwards = [round(cur_spots[j] * math.exp(-base_spreads[j] / 12.0), 5) for j in range(n_cur)]
        forward_rates.append(month_forwards)

        for _ in range(days_per_month):
            vol_state = max(0.003, min(0.025, 0.90 * vol_state + 0.10 * 0.006 + rng.gauss(0.0, 0.0007)))
            dol_shock = rng.gauss(0.0, vol_state)
            vol_shock = vol_state - 0.006
            day_rets: list[float] = []
            for j in range(n_cur):
                vol_loading = -6.0 * base_spreads[j]
                idio = rng.gauss(0.0, 0.0025)
                d_ret = dol_shock + vol_loading * vol_shock + idio
                log_spots[j] += d_ret
                day_rets.append(round(d_ret, 6))
            daily_returns.append(day_rets)

    # 1. Excess returns and forward discounts
    excess_returns: list[list[float]] = []
    forward_discounts: list[list[float]] = []
    for t in range(n_periods - 1):
        rx_row: list[float] = []
        fd_row: list[float] = []
        for j in range(n_cur):
            rx = math.log(spot_rates[t + 1][j]) - math.log(forward_rates[t][j])
            fd = math.log(forward_rates[t][j]) - math.log(spot_rates[t][j])
            rx_row.append(rx)
            fd_row.append(fd)
        excess_returns.append(rx_row)
        forward_discounts.append(fd_row)

    # 2. Sort into 5 carry portfolios
    t_ret = len(excess_returns)
    port_rets: list[list[float]] = []
    for t in range(t_ret):
        fds = forward_discounts[t]
        rxs = excess_returns[t]
        ranked_indices = sorted(range(n_cur), key=lambda j: fds[j], reverse=True)
        p_row = [0.0] * n_portfolios
        p_cnt = [0] * n_portfolios
        for rank_pos, cur_idx in enumerate(ranked_indices):
            bucket = min(int(rank_pos * n_portfolios / n_cur), n_portfolios - 1)
            p_row[bucket] += rxs[cur_idx]
            p_cnt[bucket] += 1
        port_rets.append([p_row[k] / p_cnt[k] for k in range(n_portfolios)])

    # 3. Build factors
    dol_factor = [sum(port_rets[t]) / n_portfolios for t in range(t_ret)]
    hml_factor = [port_rets[t][-1] - port_rets[t][0] for t in range(t_ret)]

    # 4. Volatility innovations
    daily_vol = [sum(abs(r) for r in row) / n_cur for row in daily_returns]
    m_months = len(daily_returns) // days_per_month
    monthly_vol = [
        sum(daily_vol[m * days_per_month : (m + 1) * days_per_month]) / days_per_month for m in range(m_months)
    ]
    y_v = monthly_vol[1:]
    x_v = [[monthly_vol[i]] for i in range(m_months - 1)]
    coeffs_v, fitted_v, _ = _multivariate_ols(x_v, y_v, fit_intercept=True)
    vol_innov = [y_v[i] - fitted_v[i] for i in range(len(y_v))]

    # 5. Lustig 2-factor model: [DOL, HML]
    t_eval = min(t_ret, len(vol_innov))
    x_lustig = [[dol_factor[t], hml_factor[t]] for t in range(t_eval)]
    mean_rets = [sum(port_rets[t][k] for t in range(t_eval)) / t_eval for k in range(n_portfolios)]

    lustig_betas: list[list[float]] = []
    for k in range(n_portfolios):
        yk = [port_rets[t][k] for t in range(t_eval)]
        c_k, _, _ = _multivariate_ols(x_lustig, yk, fit_intercept=True)
        lustig_betas.append([round(c_k[1], 3), round(c_k[2], 3)])

    _cs_lustig, _, r2_lustig = _multivariate_ols(lustig_betas, mean_rets, fit_intercept=False)

    return {
        "n_periods": n_periods,
        "hml_mean_ann": round(sum(hml_factor) / t_ret * 12.0 * 100, 2),
        "dol_mean_ann": round(sum(dol_factor) / t_ret * 12.0 * 100, 2),
        "lustig_betas": lustig_betas,
        "lustig_r2": round(r2_lustig, 3),
        "ar1_rho": round(coeffs_v[1], 3),
    }


def main() -> None:
    res = run_worked_example(seed=42)
    print("=== Rung 1 Worked Example: Currency Factor Replication (Seed 42) ===")
    print(f"Periods: {res['n_periods']} months | Annualized HML-FX Return: {res['hml_mean_ann']}%")
    print(f"Annualized Dollar Factor Return: {res['dol_mean_ann']}% | Vol AR(1) Persistence rho: {res['ar1_rho']}")
    print("Lustig Factor Betas [Beta_DOL, Beta_HML]:")
    for k, b in enumerate(res["lustig_betas"]):
        print(f"  Portfolio {k + 1}: Beta_DOL = {b[0]:>6.3f} | Beta_HML = {b[1]:>6.3f}")
    print(f"Cross-Sectional R^2: {res['lustig_r2']:.3f}")

Exemplar A · Passing level 4 (Exemplary)

Demonstrates rigorous mathematical formulation, exact Fama–MacBeth derivations, empirical horse-race analysis, and deep critique of crash risk and peso problem bias.

1. Theoretical Model and Affine SDF Representation

In no-arbitrage asset pricing, every currency excess return \(R^e_{k,t+1}\) satisfies the fundamental Euler equation \(0 = \E_t[M_{t+1} R^e_{k,t+1}]\). Following Lustig, Roussanov & Verdelhan (2011), the log stochastic discount factor in country \(j\) is affine in country-specific shocks \(z^j_{t+1}\) and common global shocks \(z^w_{t+1}\): \(-m_{t+1}^j = \lambda_j + \gamma_j z^w_{t+1} + \delta z^j_{t+1}\). When countries have heterogeneous exposures \(\gamma_j\) to global shocks, high-interest currencies have lower \(\gamma_j\) than low-interest funding currencies. Consequently, target currencies depreciate systematically during adverse global states when marginal utility is high, generating an equilibrium carry risk premium.

2. Two-Pass Fama–MacBeth Econometrics

In Stage 1, we estimate time-series multivariate OLS betas for each test portfolio \(k \in \{1,\dots,K\}\): \(R_{k,t} = a_k + \beta_{k,\mathrm{DOL}} R_{\mathrm{DOL},t} + \beta_{k,\mathrm{HML}} \mathrm{HML}_{t} + \varepsilon_{k,t}\). In Stage 2, cross-sectional factor risk premia \(\hat{\lambda}\) are estimated by regressing sample mean excess returns \(\bar{R} = (\bar{R}_1,\dots,\bar{R}_K)'\) on factor loadings \(\hat{B} = [\hat{\beta}_{\mathrm{DOL}}, \hat{\beta}_{\mathrm{HML}}]\) through the origin: \(\hat{\lambda} = (\hat{B}' \hat{B})^{-1} \hat{B}' \bar{R}\). The pricing errors are \(\hat{\alpha} = \bar{R} - \hat{B} \hat{\lambda}\), and cross-sectional fit is evaluated by \(R^2 = 1 - (\sum \hat{\alpha}_k^2) / (\sum (\bar{R}_k - \bar{\bar{R}})^2)\).

3. Factor Horse Race and Empirical Findings

On benchmark seed 42, the Lustig et al. (2011) two-factor model yields \(\beta_{\mathrm{DOL}} \approx 1.00\) across all five portfolios (level factor) and \(\beta_{\mathrm{HML}}\) spreading monotonically from \(-0.486\) (P1) to \(+0.514\) (P5) (slope factor), achieving a cross-sectional \(R^2 = 0.971\). In the Menkhoff et al. (2012) specification, global volatility innovations carry a strictly negative price of risk \(\hat{\lambda}_{\mathrm{Vol}} < 0\), while portfolio volatility betas decline monotonically from positive (P1 acts as a volatility hedge) to deeply negative (P5 suffers sharp losses during volatility spikes). This confirms that high carry returns compensate investors for bearing global foreign exchange volatility risk.

4. Structural Limitations, Crash Risk, and Peso Problems

Despite impressive sample Sharpe ratios (\(> 0.8\)), carry returns are subject to severe tail risk. As proven by Brunnermeier, Nagel & Pedersen (2008), carry trades exhibit marked negative skewness (\(\gamma_3 < -0.7\)) driven by margin-call liquidity spirals: when funding currencies appreciate, leveraged carry traders face capital depletion and fire-sale their positions, amplifying spot depreciation. Furthermore, Burnside, Eichenbaum, Kleshchelski & Rebelo (2011) demonstrate that finite-sample carry Sharpe ratios suffer from upward peso-problem bias: infrequent catastrophe devaluations that do not realize in small samples inflate historical sample means above true unconditional population returns.

  2. Cross-sectional pricing through origin: Why is the Stage 2 cross-sectional regression run without an intercept when factors are traded portfolios?
  3. Base currency invariance: If an investor uses EUR or GBP as the base currency instead of USD, does the HML-FX factor return or cross-sectional pricing change?
  4. Shanken correction: Why do standard OLS errors in Stage 2 underestimate the true standard errors of factor risk premia \(\hat{\lambda}\), and how does the Shanken (1992) correction adjust for generated regressors?
  5. Subsumption test: Does the Menkhoff et al. (2012) global FX volatility innovation factor statistically subsume the Lustig et al. (2011) HML-FX factor in spanning regressions?
  6. Volatility innovation AR(1) specification: Why must we use AR(1) residuals rather than raw volatility levels in cross-sectional asset pricing?
  7. Conditioning information: How does the forward discount sort dynamically update portfolio composition, and what is the portfolio turnover rate?
  8. Covered interest parity breakdown: How do post-2008 non-zero cross-currency bases (Du et al. 2018) impact the measurement of forward discounts and synthetic funding costs?
  9. Asymmetric liquidity spirals: What specific institutional mechanism in Brunnermeier et al. (2008) causes carry unwinds to be sharp and discontinuous while carry buildups are gradual?

  11. Crash hedging via options: Can carry crash risk be fully eliminated by purchasing out-of-the-money currency puts, and why does the volatility risk premium render naive put hedging unprofitable?
  12. Emerging vs. Developed FX: Does the HML-FX factor rely predominantly on emerging market currencies, or does it price developed G10 currencies equally well?
  13. Time-varying betas: Are currency portfolio factor loadings on global volatility constant over time, or do they spike conditionally during market turmoil?

Rung 4 · Stretch

In live currency portfolio management, execution friction and bid-ask spreads significantly erode gross factor profitability. In this extension, we simulate monthly portfolio turnover and evaluate net performance across transaction cost regimes.

Lab 3 · Transaction cost friction and breakeven spreads

Simulating dynamic monthly rebalancing with realistic turnover (approx. 0.6x to 0.8x annual turnover) will demonstrate that net HML-FX Sharpe ratios remain positive for institutional spreads under 10 bps, with a breakeven spread exceeding 200 bps.

"""Lab X2-project-3: Rung 4 Stretch: Transaction Costs & Breakeven FX Spreads.

Simulates turnover, rebalancing friction, and bid-ask spreads on HML-FX carry,
and solves for breakeven execution costs (Burnside et al. 2011, Menkhoff et al. 2012).
"""

from __future__ import annotations

import math
import random
from typing import Any


def evaluate_transaction_costs(
    n_months: int = 120,
    seed: int = 42,
) -> dict[str, Any]:
    """Evaluate gross vs net HML-FX carry returns across transaction cost tiers."""
    rng = random.Random(seed)
    n_cur = 10

    base_spreads = [-0.035, -0.020, -0.010, -0.005, 0.005, 0.010, 0.018, 0.025, 0.035, 0.055]
    cur_spreads = list(base_spreads)

    gross_hml: list[float] = []
    turnover_series: list[float] = []

    prev_p1: set[int] = set()
    prev_p5: set[int] = set()

    for _ in range(n_months):
        # Update interest spreads with noise causing occasional rank changes
        for j in range(n_cur):
            cur_spreads[j] = 0.95 * cur_spreads[j] + 0.05 * base_spreads[j] + rng.gauss(0.0, 0.003)

        # Forward discounts
        fds = [-s / 12.0 for s in cur_spreads]
        ranked = sorted(range(n_cur), key=lambda j: fds[j], reverse=True)

        cur_p1 = set(ranked[:2])  # 2 lowest rate currencies (funding)
        cur_p5 = set(ranked[-2:])  # 2 highest rate currencies (target)

        # Calculate monthly turnover fraction in long and short legs
        if prev_p1:
            turnover_long = len(cur_p5 - prev_p5) / 2.0
            turnover_short = len(cur_p1 - prev_p1) / 2.0
            # Total portfolio two-way turnover fraction
            turnover = 0.5 * (turnover_long + turnover_short)
        else:
            turnover = 1.0
        turnover_series.append(turnover)
        prev_p1 = cur_p1
        prev_p5 = cur_p5

        # Monthly return
        r_p5 = (sum(cur_spreads[j] for j in cur_p5) / 2.0) / 12.0 + rng.gauss(0.0, 0.02)
        r_p1 = (sum(cur_spreads[j] for j in cur_p1) / 2.0) / 12.0 + rng.gauss(0.0, 0.018)
        gross_hml.append(r_p5 - r_p1)

    t_obs = len(gross_hml)
    m_gross = sum(gross_hml) / t_obs
    s_gross = math.sqrt(sum((r - m_gross) ** 2 for r in gross_hml) / (t_obs - 1))
    sr_gross = (m_gross * 12.0) / (s_gross * math.sqrt(12.0))

    avg_monthly_turnover = sum(turnover_series) / t_obs
    ann_turnover = avg_monthly_turnover * 12.0

    # Net returns for various half-spread costs (in basis points: 1 bp = 0.0001)
    cost_tiers_bps = [2.0, 5.0, 10.0, 15.0, 20.0]
    tier_results: list[dict[str, float]] = []

    for c_bp in cost_tiers_bps:
        # Effective cost per rebalance = turnover * (c_bp * 1e-4 * 2 legs)
        cost_rate = c_bp * 0.0001 * 2.0
        net_hml = [r - turnover_series[t] * cost_rate for t, r in enumerate(gross_hml)]
        m_net = sum(net_hml) / t_obs
        s_net = math.sqrt(sum((r - m_net) ** 2 for r in net_hml) / (t_obs - 1))
        sr_net = (m_net * 12.0) / (s_net * math.sqrt(12.0))
        tier_results.append(
            {
                "half_spread_bps": c_bp,
                "ann_cost_pct": round(ann_turnover * cost_rate * 100, 2),
                "net_mean_pct": round(m_net * 12.0 * 100, 2),
                "net_sharpe": round(sr_net, 3),
            }
        )

    # Breakeven half-spread: solves m_gross - avg_turnover * (c * 2) = 0
    # c_breakeven = m_gross / (2 * avg_turnover)
    be_half_spread_bps = (m_gross / (2.0 * avg_monthly_turnover)) * 10000.0

    return {
        "n_months": t_obs,
        "gross_mean_ann": round(m_gross * 12.0 * 100, 2),
        "gross_vol_ann": round(s_gross * math.sqrt(12.0) * 100, 2),
        "gross_sharpe": round(sr_gross, 3),
        "ann_turnover": round(ann_turnover, 2),
        "breakeven_half_spread_bps": round(be_half_spread_bps, 1),
        "tier_results": tier_results,
    }


def main() -> None:
    res = evaluate_transaction_costs(n_months=120, seed=42)
    print("=== Rung 4 Stretch: Transaction Costs & Breakeven Spreads ===")
    mean_s = f"{res['gross_mean_ann']}%"
    vol_s = f"{res['gross_vol_ann']}%"
    print(f"Sample: {res['n_months']} months | Gross Annualized Mean: {mean_s} | Vol: {vol_s}")
    print(f"Gross Sharpe Ratio: {res['gross_sharpe']:.3f} | Annualized Turnover: {res['ann_turnover']:.2f}x")
    print("-------------------------------------------------------------------")
    print("Half-Spread | Annual Cost | Net Return (ann) | Net Sharpe Ratio")
    print("-------------------------------------------------------------------")
    for tr in res["tier_results"]:
        hs_s = f"{tr['half_spread_bps']:>4.1f} bps"
        ac_s = f"{tr['ann_cost_pct']:>5.2f}%"
        nm_s = f"{tr['net_mean_pct']:>6.2f}%"
        ns_s = f"{tr['net_sharpe']:>5.3f}"
        print(f"  {hs_s}  |   {ac_s}   |     {nm_s}    |      {ns_s}")
    print("-------------------------------------------------------------------")
    print(f"Breakeven Round-Trip / Half-Spread: {res['breakeven_half_spread_bps']:.1f} bps")


main()
