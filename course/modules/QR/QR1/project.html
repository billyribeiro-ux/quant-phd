Vintage-Aware Ingestion Engine

Quantitative research alpha decay often traces directly to historical data leakage. When a backtesting engine evaluates trading rules on modern, fully revised series or post-split prices, it assumes knowledge of macroeconomic releases and corporate restructuring that market participants did not possess. In this capstone project, you implement a production-grade, vintage-aware research data ingestion engine (vintages.py). Your engine parses and normalizes heterogeneous exchange timestamps, stores and indexes real-time data vintages in SQLite, resolves prevailing market microstructure ticks, and computes exact backward corporate action adjustment factors, proving mathematically and empirically that historical queries cannot observe revisions prior to their publication timestamps.

  • Deliverable: vintages.py passing test suite on seeds 0–49.
  • Grading: Autograded by automated pytest test harness (autograder).
  • Prerequisites: CS3 (point-in-time stores, ASOF joins), CS4 (property-based testing), S0 (corporate action breaks, data quality).

"""QR1 Project Lab 1: Rung 1 worked example of vintage-aware ingestion engine.

Demonstrates complete implementation and step-by-step verification of
point-in-time vintage indexing, UTC timestamp normalization, prevailing tick
lookups, and corporate action adjustment factor computation.
Data: SIMULATED research data store records.
"""

from __future__ import annotations

import datetime
import re
import sqlite3

_WALL_RE = re.compile(r"^(\d{4})-(\d{2})-(\d{2}) (\d{2}):(\d{2})$")


def et_to_utc(wall: str) -> str:
    """Convert Eastern wall-clock time 'YYYY-MM-DD HH:MM' to ISO 8601 UTC 'YYYY-MM-DDTHH:MM:SSZ'."""
    m = _WALL_RE.match(wall)
    if not m:
        raise ValueError(f"Malformed Eastern wall timestamp: {wall!r}")
    year, month, day, hour, minute = map(int, m.groups())
    dt = datetime.datetime(year, month, day, hour, minute)

    # US DST rule: Second Sunday of March to First Sunday of November
    d_mar1 = datetime.date(year, 3, 1)
    sun2_mar = 1 + (6 - d_mar1.weekday()) % 7 + 7
    spring_date = datetime.date(year, 3, sun2_mar)

    d_nov1 = datetime.date(year, 11, 1)
    sun1_nov = 1 + (6 - d_nov1.weekday()) % 7
    fall_date = datetime.date(year, 11, sun1_nov)

    curr_date = dt.date()
    if curr_date == spring_date and 2 <= dt.hour < 3:
        raise ValueError(f"Wall time in spring DST gap does not exist: {wall!r}")

    is_dst = (
        (spring_date < curr_date < fall_date)
        or (curr_date == spring_date and dt.hour >= 3)
        or (curr_date == fall_date and dt.hour < 2)
    )

    offset_hours = -4 if is_dst else -5
    utc_dt = dt - datetime.timedelta(hours=offset_hours)
    return utc_dt.strftime("%Y-%m-%dT%H:%M:%SZ")


def adjust_closes(
    dates: list[str],
    closes: list[float],
    events: list[tuple[str, float]],
) -> list[float]:
    """Compute backward-adjusted closes: for date d, divide by product of ratios with ex_date > d."""
    if len(dates) != len(closes):
        raise ValueError("Dates and closes length mismatch")
    for _, ratio in events:
        if ratio <= 0:
            raise ValueError(f"Split ratio must be positive: {ratio}")

    adjusted: list[float] = []
    for d, c in zip(dates, closes, strict=True):
        factor = 1.0
        for ex_date, ratio in events:
            if ex_date > d:
                factor *= ratio
        adjusted.append(c / factor)
    return adjusted


def main() -> None:
    # 1. Verify Timestamp Normalization
    wall_release = "2026-03-12 08:30"
    utc_release = et_to_utc(wall_release)
    print(f"1. Timestamp Normalization: {wall_release} ET -> {utc_release} UTC")

    # 2. In-memory Vintage Database Setup
    conn = sqlite3.connect(":memory:")
    with conn:
        conn.execute(
            """
            CREATE TABLE vintages (
                seq INTEGER PRIMARY KEY AUTOINCREMENT,
                source TEXT NOT NULL,
                series TEXT NOT NULL,
                obs_date TEXT NOT NULL,
                value TEXT NOT NULL,
                published TEXT NOT NULL,
                retrieved TEXT NOT NULL
            )
            """
        )
        conn.execute("CREATE INDEX idx_vintages ON vintages(source, series, obs_date, published, seq)")

    # 3. Ingest Vintages (Initial advance release, second estimate revision, third estimate revision)
    source = "fred"
    series = "GDP_GROWTH"
    obs_date = "2025-Q4"

    records = [
        (source, series, obs_date, "2.1", "2026-01-29T13:30:00Z", "2026-01-29T13:35:00Z"),
        (source, series, obs_date, "2.4", "2026-02-26T13:30:00Z", "2026-02-26T13:35:00Z"),
        (source, series, obs_date, "2.3", "2026-03-26T13:30:00Z", "2026-03-26T13:35:00Z"),
    ]

    with conn:
        conn.executemany(
            "INSERT INTO vintages (source, series, obs_date, value, published, retrieved) VALUES (?, ?, ?, ?, ?, ?)",
            records,
        )

    print("\n2. Ingested Macro Vintages:")
    for r in records:
        print(f"  Obs {r[2]} published={r[4]} val={r[3]}%")

    # 4. Point-in-time queries
    def query_visible(t: str) -> str | None:
        cur = conn.cursor()
        cur.execute(
            """
            SELECT value FROM vintages
            WHERE source = ? AND series = ? AND obs_date = ? AND published <= ?
            ORDER BY published DESC, seq DESC LIMIT 1
            """,
            (source, series, obs_date, t),
        )
        row = cur.fetchone()
        return row[0] if row else None

    cutoffs = [
        ("2026-01-15T00:00:00Z", "Prior to any publication"),
        ("2026-02-15T00:00:00Z", "Post-Advance release"),
        ("2026-03-15T00:00:00Z", "Post-Second release"),
        ("2026-04-15T00:00:00Z", "Post-Third release"),
    ]

    print("\n3. Point-in-Time Cut-off Queries:")
    for cut_t, label in cutoffs:
        val = query_visible(cut_t)
        print(f"  Cut-off {cut_t} [{label}]: visible={val}")

    # 5. Split Adjustment Check
    dates = ["2026-03-02", "2026-03-03", "2026-03-04", "2026-03-05"]
    closes = [100.0, 100.0, 50.0, 52.0]  # 2:1 split on 2026-03-04
    events = [("2026-03-04", 2.0)]
    adj = adjust_closes(dates, closes, events)

    print("\n4. Corporate Action Adjustment:")
    for d, raw, a in zip(dates, closes, adj, strict=True):
        print(f"  {d}: raw={raw:.1f} adjusted={a:.1f}")

    conn.close()
    print("\nRung 1 worked example completed successfully.")

"""QR1 Project Lab 3: Rung 3 independent project deliverable verification.

from __future__ import annotations

import datetime
import re
import sqlite3

_WALL_RE = re.compile(r"^(\d{4})-(\d{2})-(\d{2}) (\d{2}):(\d{2})$")


def et_to_utc(wall: str) -> str:
    m = _WALL_RE.match(wall)
    if not m:
        raise ValueError(f"Malformed Eastern wall timestamp: {wall!r}")
    y, mth, d, h, mn = map(int, m.groups())
    dt = datetime.datetime(y, mth, d, h, mn)

    d_mar1 = datetime.date(y, 3, 1)
    sun2_mar = 1 + (6 - d_mar1.weekday()) % 7 + 7
    spring_date = datetime.date(y, 3, sun2_mar)

    d_nov1 = datetime.date(y, 11, 1)
    sun1_nov = 1 + (6 - d_nov1.weekday()) % 7
    fall_date = datetime.date(y, 11, sun1_nov)

    cdate = dt.date()
    if cdate == spring_date and 2 <= dt.hour < 3:
        raise ValueError("Spring DST gap does not exist")

    is_dst = (
        (spring_date < cdate < fall_date)
        or (cdate == spring_date and dt.hour >= 3)
        or (cdate == fall_date and dt.hour < 2)
    )
    offset = datetime.timedelta(hours=-4 if is_dst else -5)
    return (dt - offset).strftime("%Y-%m-%dT%H:%M:%SZ")


def last_trade_at(ticks: list[tuple[str, int, float]], t: str) -> float | None:
    cand = [tick for tick in ticks if tick[0] <= t]
    if not cand:
        return None
    return max(cand, key=lambda x: (x[0], x[1]))[2]


def adjust_closes(dates: list[str], closes: list[float], events: list[tuple[str, float]]) -> list[float]:
    if len(dates) != len(closes):
        raise ValueError("length mismatch")
    for _, r in events:
        if r <= 0:
            raise ValueError("ratio must be positive")
    res: list[float] = []
    for d, c in zip(dates, closes, strict=True):
        f = 1.0
        for ed, r in events:
            if ed > d:
                f *= r
        res.append(c / f)
    return res

Specification

Context
Quantitative asset management firms execute backtests across multi-decade spans containing hundreds of thousands of corporate action adjustments, economic revisions, and microstructure order stream ticks. A strategy backtest must never observe information prior to its real-world publication timestamp, and capital changes like stock splits must never generate artificial return breaks. You are tasked with developing vintages.py, extending the research data infrastructure to enforce point-in-time integrity across macro vintages, tick feeds, and corporate actions.
Inputs
SIMULATED fixtures generated deterministically per workspace seed: macroeconomic vintage tables from the five canonical sources (fred, french, shiller, treasury, edgar), tick stream tuples with timestamp jitter, daily stock closing prices, and split event tuples.
Required outputs
A single Python module vintages.py exposing the constant SOURCES: tuple[str, ...] = ("fred", "french", "shiller", "treasury", "edgar") and implementing:
  • et_to_utc(wall: str) -> str: converts YYYY-MM-DD HH:MM in America/New_York to canonical ISO 8601 UTC YYYY-MM-DDTHH:MM:SSZ. Rejects spring DST gap (02:00–02:59) and resolves autumn overlap (01:00–01:59) to earlier instant (daylight time, UTC−4).
  • class VintageIndex(path: str): SQLite-backed store with methods:
    • record(source: str, series: str, obs_date: str, value: str, published: str, retrieved: str) -> None: inserts vintage; idempotent on exact duplicate; raises on conflicting value or invalid timestamp/source.
    • visible(source: str, series: str, obs_date: str, t: str) -> str | None: returns latest value published at or before cut-off t.
    • visible_series(source: str, series: str, t: str) -> dict[str, str]: returns mapping of obs_date to visible value for all dates having a visible vintage at t.
    • close() -> None: closes database connection.
  • last_trade_at(ticks: list[tuple[str, int, float]], t: str) -> float | None: resolves prevailing trade price maximizing (timestamp, seq) among ticks with timestamp <= t.
  • adjust_closes(dates: list[str], closes: list[float], events: list[tuple[str, float]]) -> list[float]: backward-adjusts daily close prices for stock splits, dividing each close by the product of split ratios for events with ex_date > d.
Constraints
Standard library only (datetime, sqlite3, re, bisect, math). No network access, no DuckDB dependency, and no third-party libraries. All timestamps must strictly follow ISO 8601 UTC with trailing Z. Clean execution under python -I -W error without leaking database connections.
Grading
Autograder. The project is graded by an automated pytest test harness running locally and in Pyodide across 50 independent seeds (0–49), evaluating property-based invariants, mathematical split return preservation, timestamp boundary conditions, and exception contracts.
