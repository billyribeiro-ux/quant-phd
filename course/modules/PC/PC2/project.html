"""PC2 Lab Project-1: Rung 1 worked example: equity pitch and IC adversarial defense validator.

SIMULATED. Validates structural and quantitative integrity of a long equity pitch
derived from an A16 discounted cash flow model, together with its 5-part scripted IC defense.
Standard library only.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True)
class EquityPitchModel:
    ticker: str
    current_price: float
    base_target: float
    bull_target: float
    bear_target: float
    base_prob: float
    bull_prob: float
    bear_prob: float
    catalysts: list[str]
    premortem_tripwires: list[str]
    sizing_weight_nav: float
    ic_defense_dialogues: list[tuple[str, str, str]]  # (Attack Vector, Question, 3-Step Defense)


def validate_equity_pitch(pitch: EquityPitchModel) -> dict[str, bool | float | str]:
    """Validate completeness and financial metrics of an institutional equity pitch."""
    prob_sum = pitch.base_prob + pitch.bull_prob + pitch.bear_prob
    probs_valid = abs(prob_sum - 1.0) < 1e-6

    exp_price = (
        pitch.base_prob * pitch.base_target + pitch.bull_prob * pitch.bull_target + pitch.bear_prob * pitch.bear_target
    )
    exp_return = (exp_price - pitch.current_price) / pitch.current_price

    upside = (pitch.base_target - pitch.current_price) / pitch.current_price
    downside = (pitch.current_price - pitch.bear_target) / pitch.current_price
    asymmetry = upside / downside if downside > 0 else 0.0

    has_catalysts = len(pitch.catalysts) >= 2
    has_premortem = len(pitch.premortem_tripwires) >= 3
    has_ic_defense = len(pitch.ic_defense_dialogues) >= 5

    # Check 3-step defense structure in every dialogue: must contain acknowledge, evidence, invalidation
    valid_defense_steps = True
    for _vector, _q, defense in pitch.ic_defense_dialogues:
        has_ack = "acknowledge" in defense.lower() or "concede" in defense.lower() or "bound" in defense.lower()
        has_evid = "evidence" in defense.lower() or "historical" in defense.lower() or "data" in defense.lower()
        has_exit = "tripwire" in defense.lower() or "exit" in defense.lower() or "liquidate" in defense.lower()
        if not (has_ack and has_evid and has_exit):
            valid_defense_steps = False

    is_compliant = (
        probs_valid
        and asymmetry >= 2.0
        and exp_return >= 0.15
        and has_catalysts
        and has_premortem
        and has_ic_defense
        and valid_defense_steps
    )

    return {
        "probs_valid": probs_valid,
        "expected_price": exp_price,
        "expected_return": exp_return,
        "asymmetry_ratio": asymmetry,
        "has_catalysts": has_catalysts,
        "has_premortem": has_premortem,
        "has_ic_defense": has_ic_defense,
        "valid_defense_steps": valid_defense_steps,
        "is_compliant": is_compliant,
    }

def evaluate_macro_pitch(pitch: CurveSteepenerPitch) -> dict[str, bool | float]:
    """Verify quantitative parameters and IC defense completeness of macro curve pitch."""
    # Compute basis point payoff asymmetry
    upside_bps = pitch.base_target_bps - pitch.current_spread_bps
    downside_bps = pitch.current_spread_bps - pitch.bear_target_bps
    asymmetry = upside_bps / downside_bps if downside_bps > 0 else 0.0

    expected_bps = 0.55 * pitch.base_target_bps + 0.20 * pitch.bull_target_bps + 0.25 * pitch.bear_target_bps
    expected_gain_bps = expected_bps - pitch.current_spread_bps

    # Verify DV01 hedge ratio consistency: 2Y contracts per 10Y contract = dv01_10y / dv01_2y
    expected_ratio = pitch.dv01_10y / pitch.dv01_2y
    ratio_valid = abs(pitch.dv01_hedge_ratio - expected_ratio) < 1e-4

    has_catalysts = len(pitch.catalysts) >= 2
    has_tripwires = len(pitch.premortem_tripwires) >= 3
    has_ic_defense = len(pitch.ic_defense_dialogues) >= 5

    valid_defense_steps = True
    for _vec, _q, defense in pitch.ic_defense_dialogues:
        has_ack = "acknowledge" in defense.lower() or "concede" in defense.lower() or "bound" in defense.lower()

    is_compliant = (
        ratio_valid
        and asymmetry >= 2.0
        and expected_gain_bps >= 25.0
        and has_catalysts
        and has_tripwires
        and has_ic_defense
        and valid_defense_steps

def verify_pitch_submission(p: PitchSubmission) -> dict[str, bool]:

    has_asymmetry = p.asymmetry_ratio >= 2.0
    has_gain = p.expected_gain_pct >= 10.0
    has_catalysts = p.catalyst_count >= 2
    has_premortem = p.premortem_tripwire_count >= 3

    # Check 3-step defense structure in all dialogues
    dialogues_structured = True
    for _vec, _q, ans in p.ic_dialogues:
        has_ack = "acknowledge" in ans.lower() or "bound" in ans.lower() or "concede" in ans.lower()

Grading Rubric

The project is evaluated across five observable criteria C1–C5, each defining four distinct performance levels:

Criterion Level 4 (Exemplary) Level 3 (Proficient) Level 2 (Developing) Level 1 (Unacceptable)
C1 · Variant Perception & Thesis Explicitly quantifies divergence between consensus estimates and proprietary model with exact percentage or basis point deltas; names the precise economic mechanism causing market mispricing. States a clear variant perception contrasting with consensus, but lacks exact quantitative gap metrics or omits one underlying economic driver. States a qualitative investment thesis without benchmarking against sell-side consensus or quantifying market divergence. Presents generic descriptive company or macro background without any identifiable variant perception.
C2 · Valuation & Asymmetry Modeling Models full Base, Bull, and Bear scenarios with articulated DCF or spread parameters; demonstrates asymmetry ratio \(A \ge 2.0x\) and calculates margin of safety \(\ge 15\%\). Models three scenarios with \(A \ge 2.0x\), but scenario probabilities or cash flow driver assumptions lack explicit mathematical articulation. Provides only a single target price or models symmetric upside and downside (\(A < 2.0x\)), failing institutional asymmetry hurdles. Relies on an arbitrary valuation multiple without DCF cash flow modeling, scenario distributions, or downside stress testing.
C3 · Catalysts & Timing Horizon Identifies at least two discrete, observable calendar catalysts with explicit dates within 30 to 180 days; details the exact pricing transmission mechanism forcing market re-rating. Identifies two catalysts with calendar dates, but describes the market repricing mechanism vaguely or includes one non-observable event. Lists only one catalyst, or relies on vague, non-time-bounded assertions such as "management will execute on growth." Omits catalysts entirely, assuming that market pricing efficiency will correct valuation gaps without any resolving event.
C4 · Pre-Mortem & Invalidation Tripwires Executes Gary Klein prospective hindsight protocol across at least three risk vectors; establishes non-negotiable quantitative tripwires that trigger mandatory liquidation. Conducts pre-mortem analysis with three failure modes, but one tripwire relies on subjective qualitative assessment rather than an observable metric. Lists standard risk caveats without prospective hindsight framing, or sets vague stop thresholds without automated liquidation directives. Omits pre-mortem analysis or asserts that the investment thesis is free of operational and downside risks.
C5 · Risk Sizing & IC Adversarial Defense Sizes position via fractional Kelly and Almgren-Chriss liquidity horizons; defends five distinct IC attacks using the rigorous 3-step protocol (Acknowledge, Evidence, Tripwire). Derives fractional Kelly sizing and defends five IC attacks, but one defense response lacks an empirical base rate or fails to state an exit tripwire. Sizing is arbitrary without Kelly or liquidity calculations; defenses are reactive, defensive, or evade committee challenges. Exceeds fund risk policy caps or liquidity exit limits; defenses are combative, dismissive, or completely unscripted.

Investment Committee Question Bank

Your scripted defenses must select five questions for your equity pitch and five questions for your macro pitch from this 16-angle institutional interrogation bank:

  1. Terminal Value Dominance: "Your DCF model assigns over 70% of Enterprise Value to the terminal value. If long-term benchmark discount rates rise by 75 bps, how can we deploy capital when the entire margin of safety is erased?"
  2. Hedge Fund Crowding & Float Concentration: "Top-10 institutional hedge funds hold over 35% of the free float. If factor momentum unwinds, won't correlated liquidation cascades drive the asset far past your Bear stop-loss?"
  3. Competitive Feature Bundling: "Cloud hyperscalers or dominant platform incumbents are bundling competing features into their core enterprise licenses. Why won't gross margins compress by 500+ basis points?"
  4. Execution Slippage & Sales Inefficiency: "Customer acquisition costs increased significantly last quarter while sales productivity decelerated. What quantitative proof confirms that marketing expansion will not consume operating liquidity?"
  5. Customer Concentration Vulnerability: "The top three enterprise clients generate over 30% of total recurring revenue. If a single customer consolidates or renegotiates contract terms, does the equity value collapse below debt value?"
  6. Balance Sheet Refinancing Cliff: "The firm faces a major debt maturity wall in 18 months during an elevated credit spread regime. How will refinancing interest expense impact free cash flow to equity?"
  7. Forensic Accounting Red Flags: "Days Sales in Receivables Index (DSRI) exceeds 1.20 and operating cash flow conversion has fallen below 60% of GAAP operating income. How do you prove earnings are not being accelerated?"
  8. Management Capital Allocation Track Record: "Executive management has completed multiple dilutive acquisitions with declining return on invested capital. Why should the committee trust their forward Capex guidance?"
  9. Macro IT Budget Freezes: "Corporate CIO surveys indicate enterprise software budget growth is decelerating to zero. What empirical proof demonstrates that target contracts are immune to corporate budget cuts?"
  10. Negative Carry & Roll Drag: "The yield curve is deeply inverted, bleeding significant carry every month. How long can the fund hold this position before cumulative financing costs consume the expected alpha?"
  11. Duration Hedge Instability (Convexity Drift): "Your duration-neutral contract ratio was calculated at spot yields. During a 100 bps parallel rate shift, won't negative convexity drift unhedge your DV01 exposure?"
  12. Foreign Central Bank Policy Spillovers: "If the ECB or Bank of Japan executes unexpected policy rate changes, European or Asian bond yields could distort domestic yield curves. How does your model isolate domestic monetary drivers?"
  13. Sticky Wage Inflation & Rate Path Deferral: "Service inflation and labor market metrics remain resilient. What if central bank policy rate easing is deferred by an additional 12 months?"
  14. Emergency Liquidity Evaporation: "Average daily volume measured during bull markets contracts by 60% during a liquidity panic. In a crisis, how will you liquidate within your mandated days without 20% price slippage?"
  15. Regulatory & Antitrust Intervention: "Antitrust authorities are scrutinizing platform cross-subsidization and common ownership. What regulatory rulings could invalidate the firm's pricing power?"
  16. Supply Chain Bottlenecks & Inventory Aging: "Component lead times are lengthening and raw material inventory days have expanded by 25%. How will inventory write-downs affect gross margins over the coming two quarters?"
