Rungs 1 and 2

Lab 1 · Worked example: VAR variance decomposition pipeline

Estimates a first-order VAR on simulated stock returns and dividend-price ratios, derives news multiplier vectors, decomposes return innovations, and verifies long-horizon variance shares.

"""Lab AP5-project-1 (Rung 1, Worked Example): Fully Solved VAR Variance Decomposition.

SIMULATED: complete end-to-end Campbell-Shiller VAR variance decomposition pipeline
estimating return innovations, cash-flow news, discount-rate news, and long-horizon shares.
"""

from __future__ import annotations

import numpy as np


def main() -> None:
    rng = np.random.default_rng(42)
    n_obs = 200
    rho = 0.96

    # 1. Simulate calibrated return and dividend-price system
    # z_t = [r_t, dp_t]^T
    phi_true = np.array([[0.00, 0.10], [0.00, 0.93]])
    c_true = np.array([0.08, -3.20 * (1.0 - 0.93)])
    sigma_u = 0.18
    sigma_v = 0.14
    cov_uv = -0.72 * sigma_u * sigma_v
    cov_matrix = np.array([[sigma_u**2, cov_uv], [cov_uv, sigma_v**2]])

    eps = rng.multivariate_normal(mean=[0.0, 0.0], cov=cov_matrix, size=n_obs + 1)
    z = np.zeros((n_obs + 1, 2))
    z[0] = [0.08, -3.20]
    for t in range(n_obs):
        z[t + 1] = c_true + phi_true @ z[t] + eps[t + 1]

    # 2. OLS VAR(1) Estimation
    z_lag = z[:-1]
    z_lead = z[1:]
    x_mat = np.column_stack([np.ones(n_obs), z_lag])
    b_mat = np.linalg.lstsq(x_mat, z_lead, rcond=None)[0]
    phi_hat = b_mat[1:].T
    residuals = z_lead - x_mat @ b_mat
    sigma_hat = (residuals.T @ residuals) / (n_obs - 3)

    # 3. Campbell-Shiller News Vectors
    e_r = np.array([1.0, 0.0])
    eye_2 = np.eye(2)
    inv_mat = np.linalg.inv(eye_2 - rho * phi_hat)

    # lambda_dr = e_r^T rho Phi (I - rho Phi)^(-1)
    lambda_dr = e_r @ (rho * phi_hat) @ inv_mat
    # lambda_cf = e_r^T (I - rho Phi)^(-1)
    lambda_cf = e_r @ inv_mat

    n_dr = residuals @ lambda_dr
    n_cf = residuals @ lambda_cf
    eta_r = residuals[:, 0]

    # Check return innovation identity: eta_r = n_cf - n_dr
    max_id_err = float(np.max(np.abs(eta_r - (n_cf - n_dr))))

    # 4. Return Innovation Variance Decomposition
    var_eta = float(np.var(eta_r, ddof=1))
    var_cf = float(np.var(n_cf, ddof=1))
    var_dr = float(np.var(n_dr, ddof=1))
    cov_cf_dr = float(np.cov(n_cf, n_dr, ddof=1)[0, 1])

    share_dr = var_dr / var_eta
    share_cf = var_cf / var_eta
    share_cov = -2.0 * cov_cf_dr / var_eta

    # 5. Dividend-Yield Variance Decomposition
    # Solve discrete Lyapunov equation for population state covariance
    eye_4 = np.eye(4)
    gamma_vec = np.linalg.inv(eye_4 - np.kron(phi_hat, phi_hat)) @ sigma_hat.flatten()
    gamma_0 = gamma_vec.reshape((2, 2))

    e_dp = np.array([0.0, 1.0])
    long_run_r_op = inv_mat @ phi_hat
    long_run_d_op = (e_r @ phi_hat + rho * e_dp @ phi_hat - e_dp) @ inv_mat

    b_r = float((e_r @ long_run_r_op @ gamma_0 @ e_dp) / (e_dp @ gamma_0 @ e_dp))
    b_d = float((long_run_d_op @ gamma_0 @ e_dp) / (e_dp @ gamma_0 @ e_dp))
    identity_b = b_r - b_d

    print("Project Rung 1: VAR Variance Decomposition Worked Example")
    print(f"Sample length T:             {n_obs}")
    print(f"Estimated rho:               {rho:.4f}")
    print(f"VAR Phi[0,1] (return slope): {phi_hat[0, 1]:.4f}")
    print(f"VAR Phi[1,1] (persistence):  {phi_hat[1, 1]:.4f}")
    print(f"Max return news discrepancy: {max_id_err:.8e}")
    print(f"Unexpected return var:       {var_eta:.6f}")
    print(f"Discount-rate news var:      {var_dr:.6f} ({share_dr * 100:.2f}%)")
    print(f"Cash-flow news var:          {var_cf:.6f} ({share_cf * 100:.2f}%)")
    print(f"News covariance share:       {share_cov * 100:.2f}%")
    print(f"Long-horizon return share b_r:   {b_r * 100:.2f}%")
    print(f"Long-horizon dividend share b_d: {b_d * 100:.2f}%")
    print(f"Identity b_r - b_d:              {identity_b * 100:.2f}%")

"""Lab AP5-project-3 (Rung 3, Reference Run): Benchmark Outputs for Autograder Validation.

SIMULATED: runs the reference implementation on fixed benchmark inputs to establish
gold-standard test numbers for independent project verification.
"""

from __future__ import annotations

import numpy as np


def main() -> None:
    # Fixed benchmark input matrix: 100 periods of [r_t, dp_t]
    rng = np.random.default_rng(2024)
    n_obs = 100
    rho = 0.96

    phi_bench = np.array([[0.01, 0.12], [-0.02, 0.91]])
    sigma_bench = np.array([[0.035, -0.018], [-0.018, 0.020]])
    c_bench = np.array([0.07, -0.28])

    eps = rng.multivariate_normal(mean=[0.0, 0.0], cov=sigma_bench, size=n_obs + 1)
    z = np.zeros((n_obs + 1, 2))
    z[0] = [0.07, -3.10]
    for t in range(n_obs):
        z[t + 1] = c_bench + phi_bench @ z[t] + eps[t + 1]

    # 1. Estimate VAR
    z_lag = z[:-1]
    z_lead = z[1:]
    x_mat = np.column_stack([np.ones(n_obs), z_lag])
    b_mat = np.linalg.lstsq(x_mat, z_lead, rcond=None)[0]
    c_hat = b_mat[0]
    phi_hat = b_mat[1:].T
    resids = z_lead - x_mat @ b_mat
    sigma_hat = (resids.T @ resids) / (n_obs - 3)

    # 2. Campbell-Shiller News
    e_r = np.array([1.0, 0.0])
    eye_2 = np.eye(2)
    inv_mat = np.linalg.inv(eye_2 - rho * phi_hat)
    lambda_dr = e_r @ (rho * phi_hat) @ inv_mat
    lambda_cf = e_r @ inv_mat

    n_dr = resids @ lambda_dr
    n_cf = resids @ lambda_cf
    eta = resids[:, 0]

    # Variances
    var_eta = float(np.var(eta, ddof=1))
    var_dr = float(np.var(n_dr, ddof=1))
    var_cf = float(np.var(n_cf, ddof=1))
    cov_news = float(np.cov(n_cf, n_dr, ddof=1)[0, 1])

    # 3. Dividend-yield variance shares
    eye_4 = np.eye(4)
    gamma_vec = np.linalg.inv(eye_4 - np.kron(phi_hat, phi_hat)) @ sigma_hat.flatten()
    gamma_0 = gamma_vec.reshape((2, 2))

    e_dp = np.array([0.0, 1.0])
    long_run_r = inv_mat @ phi_hat
    long_run_d = (e_r @ phi_hat + rho * e_dp @ phi_hat - e_dp) @ inv_mat

    b_r = float((e_r @ long_run_r @ gamma_0 @ e_dp) / (e_dp @ gamma_0 @ e_dp))
    b_d = float((long_run_d @ gamma_0 @ e_dp) / (e_dp @ gamma_0 @ e_dp))

    # 4. Stambaugh bias for predictive regression of r on dp
    phi_dp = phi_hat[1, 1]
    sigma_uv = float(sigma_hat[0, 1])
    sigma_v2 = float(sigma_hat[1, 1])

    print(f"Return innovation var:       {var_eta:.6f}")
    print(f"Discount-rate news var:      {var_dr:.6f}")
    print(f"Cash-flow news var:          {var_cf:.6f}")

Specification: Variance Decomposition Engine

Context
Asset allocators and macroeconomic researchers need to understand whether swings in aggregate equity market valuations reflect changing expectations of corporate dividend growth (cash-flow fundamentals) or changing discount rates (time-varying equity risk premia). You will implement a complete, robust VAR variance decomposition pipeline in Python, validate it with the autograder, and synthesize your empirical findings in a research note.
Inputs
Multivariate time series of log equity returns \(r_t\), log dividend growth \(\Delta d_t\), and log dividend-price ratios \(dp_t\), generated deterministically by grader/instance.py across 50 evaluation seeds.
Required outputs
A Python module implementing the variance decomposition engine exported by grader/reference.py: 
# Required API signatures:
estimate_var(z: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]
campbell_shiller_news(phi: np.ndarray, residuals: np.ndarray, rho: float = 0.96) -> tuple[np.ndarray, np.ndarray]
decompose_return_variance(n_cf: np.ndarray, n_dr: np.ndarray) -> dict[str, float]
dividend_yield_variance_decomposition(phi: np.ndarray, sigma: np.ndarray, rho: float = 0.96) -> dict[str, float]
predictive_regression(x: np.ndarray, y: np.ndarray, max_lags: int = 4) -> dict[str, float]
stambaugh_bias(phi_x: float, sigma_uv: float, sigma_v2: float, n_obs: int) -> float
out_of_sample_r2(y: np.ndarray, x: np.ndarray, n_init: int = 60, constrained: bool = False) -> float
and a 4-page research note evaluating sample sensitivity, parameter stability, and macroeconomic implications.
Constraints
Standard library and NumPy only; no SciPy, no statsmodels. All functions must be deterministic, validate input dimensions, raise ValueError on invalid inputs, and execute in under 5 seconds per seed in Pyodide.
Grading
Graded by both autograder and rubric. The autograder tests mathematical identities and numerical accuracy across 50 consecutive seeds, while the rubric evaluates the accompanying empirical research note across five criteria.

"We estimate a first-order bivariate vector autoregression on annual CRSP log real returns \(r_{t+1}\) and log dividend-price ratios \(dp_{t+1}\). The estimated companion matrix \(\hat\Phi\) exhibits eigenvalues \(\lambda_1 = 0.938\) and \(\lambda_2 = 0.015\), strictly satisfying stationarity. Using the Campbell-Shiller linearization parameter \(\rho = 0.96\), the return innovation identity \(r_{t+1} - \mathbb{E}_t[r_{t+1}] = N_{CF, t+1} - N_{DR, t+1}\) holds to a maximum absolute error of \(1.1 \times 10^{-16}\). Decomposing the unexpected return variance \(\operatorname{Var}(\eta_r) = 0.0302\), discount-rate news accounts for 74.2% of the variance, cash-flow news accounts for 42.1%, and their covariance contributes \(-16.3\%\). Solving the discrete Lyapunov equation \(\operatorname{vec}(\Gamma_0) = (I - \Phi \otimes \Phi)^{-1} \operatorname{vec}(\Sigma)\), the population dividend-yield variance shares are \(b_r = 0.985\) and \(b_d = -0.015\), satisfying \(b_r - b_d = 1.000000000000\). In predictive regressions, the naive OLS slope \(\hat\beta = 0.098\) (\(t_{NW} = 2.14\)) is subject to a positive Stambaugh small-sample bias of \(0.034\) due to the strong negative correlation \(\rho(u, v) = -0.73\) and persistence \(\phi = 0.938\). The bias-corrected slope is \(\hat\beta_{adj} = 0.064\) (\(t_{adj} = 1.40\)). Recursive out-of-sample forecasting confirms the Welch-Goyal critique for unconstrained OLS (\(R^2_{OOS} = -0.42\%\)), whereas imposing Campbell-Thompson economic non-negativity restrictions restores positive out-of-sample predictive power (\(R^2_{OOS} = +1.85\%\)), generating an annualized certainty equivalent utility gain of 48 basis points for a mean-variance investor with risk aversion \(\gamma = 3\)."
