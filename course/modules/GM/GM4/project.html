Atlas A20: Macro Trade-Construction Engine

Quantitative macro investing requires a deterministic bridge between fundamental economic views and production trading portfolios. In this milestone project, you build Atlas A20 (atlas/strategy.py): an institutional macro trade-construction engine that transforms discrete macroeconomic views into duration-neutral relative-value spreads, analyzes option versus delta-one payoff convexity, attributes holding carry and repo financing drag, sizes capital via fractional Kelly rules under leverage bounds, and evaluates full non-linear scenario P&L grids across historical stress regimes.

  • Deliverable: atlas/strategy.py passing acceptance test suite on seeds 0–49.
  • Grading: Autograded by automated pytest test harness (autograder).
  • Milestone: Atlas A20 (contract specification).

Project Specification

Context
Institutional macro funds manage multi-asset portfolios spanning sovereign yield curves, foreign exchange, equities, and derivative volatility. Atlas A20 establishes the definitive execution architecture for Milestone A20, integrating yield curve bootstrapping from A5, analytical option pricing from A6, portfolio risk budgeting from A11, and factor stress testing from A12 into an end-to-end strategy construction pipeline.
Inputs
Structured MacroView instances specifying asset IDs, view types (steepener, flattener, directional, volatility), expected basis-point moves, and confidence levels; yield curve tenor specifications with DV01 sensitivities, spot yields, and duration parameters; option surface vectors (strikes, implied volatilities, maturities, and position weights); and 2D macroeconomic shock grids (spot and volatility shift vectors).
Required Outputs
A compliant Python module atlas/strategy.py implementing the core contract functions: compute_dv01_hedge_ratio, construct_curve_spread, evaluate_carry_and_rolldown, compute_expression_convexity, compute_asymmetry_ratio, size_trade_fractional_kelly, scenario_pnl_grid, and the coordinating class MacroTradeEngine.
Constraints
Exact duration neutrality on curve spreads (\(|\text{net DV01}| < 10^{-4}\)); analytical BSM Greek consistency; numerical non-linear revaluation on scenario grids rather than linear approximations; fractional Kelly leverage bounds clamped to maximum gross leverage limits; strict adherence to PEP 8 standards with full type annotations passing mypy --strict and ruff.
Grading
Autograded by automated test harness suite (grader/test_project.py) across 50 independent pseudorandom seeds (0–49), accompanied by mutation testing killing 5 targeted implementation mutants.

Lab Project-1 · Atlas A20 Reference Worked Example

Demonstrates end-to-end execution of a 2s10s curve steepener trade, calculating exact DV01 hedge ratios, carry attribution, and quarter-Kelly sizing.

"""Lab GM4-project-1: Rung 1 Worked Example: End-to-End Atlas A20 Engine Pipeline.

SIMULATED. Complete worked instance demonstrating MacroView specification,
DV01-neutral curve hedge ratio calculation, carry attribution, and scenario stress revaluation.
"""

from __future__ import annotations

from typing import Any


def run_worked_example() -> dict[str, Any]:
    # 1. View Definition: 25 bps steepening expected over 90 days with 75% confidence
    view_move_bps = 25.0
    confidence = 0.75
    horizon_years = 90.0 / 365.0

    # 2. Leg Parameters: 2Y Note vs 10Y Note
    dv01_2y = 42.50  # per $1M
    dv01_10y = 85.00  # per $1M
    yield_2y = 0.042
    yield_10y = 0.046
    dur_2y = 1.95
    dur_10y = 8.10
    roll_bps_2y = 6.0
    roll_bps_10y = 16.0
    repo_rate = 0.038

    # 3. Hedge Ratio and Sizing: $10M front-end notional
    notional_2y = 10_000_000.0
    hedge_ratio = dv01_2y / dv01_10y  # 0.50
    notional_10y = -notional_2y * hedge_ratio  # -$5M short 10Y

    net_dv01 = (notional_2y / 1e6) * dv01_2y + (notional_10y / 1e6) * dv01_10y

    # 4. Carry & Roll-Down Attribution
    gross_carry_2y = yield_2y * horizon_years
    roll_2y = (roll_bps_2y * 1e-4) * dur_2y * horizon_years
    drag_2y = repo_rate * horizon_years
    net_carry_2y_bps = (gross_carry_2y + roll_2y - drag_2y) * 10_000.0

    gross_carry_10y = yield_10y * horizon_years
    roll_10y = (roll_bps_10y * 1e-4) * dur_10y * horizon_years
    drag_10y = repo_rate * horizon_years
    net_carry_10y_bps = (gross_carry_10y + roll_10y - drag_10y) * 10_000.0

    # Portfolio net carry: weighted by signed notional
    w_2y = 10.0 / 15.0
    w_10y = -5.0 / 15.0
    port_carry_bps = w_2y * net_carry_2y_bps + w_10y * net_carry_10y_bps

    # 5. Kelly Sizing
    expected_return = (view_move_bps * 1e-4) * confidence
    vol_est = 0.065
    var_est = vol_est**2
    quarter_kelly = 0.25 * (expected_return / var_est)
    capped_kelly = min(4.0, max(-4.0, quarter_kelly))

    return {
        "hedge_ratio": hedge_ratio,
        "notional_2y": notional_2y,
        "notional_10y": notional_10y,
        "net_dv01": net_dv01,
        "carry_2y_bps": net_carry_2y_bps,
        "carry_10y_bps": net_carry_10y_bps,
        "portfolio_net_carry_bps": port_carry_bps,
        "quarter_kelly_weight": capped_kelly,
    }


def main() -> None:
    res = run_worked_example()
    print("=== Atlas A20 Worked Example: 2s10s Steepener ===")
    print(f"DV01 Hedge Ratio (10Y per 2Y): {res['hedge_ratio']:.4f}")
    print(f"Front-End (2Y) Notional: ${res['notional_2y']:,.0f}")
    print(f"Back-End (10Y) Notional: ${res['notional_10y']:,.0f}")
    print(f"Net Portfolio DV01 ($): {res['net_dv01']:.2f}")
    print(f"2Y Net Carry: {res['carry_2y_bps']:.1f} bps")
    print(f"10Y Net Carry: {res['carry_10y_bps']:.1f} bps")
    print(f"Portfolio Net Carry: {res['portfolio_net_carry_bps']:.1f} bps")
    print(f"Quarter Kelly Leverage Weight: {res['quarter_kelly_weight']:.2f}x")


main()

Lab Project-3 · Dynamic Cross-Asset Hedging and Market Impact

Simulates dynamic delta rebalancing of a short option portfolio under stochastic underlying asset jumps and power-law temporary price impact.

"""Lab GM4-project-3: Rung 4 Stretch Goal: Dynamic Cross-Asset Macro Rebalancing and Liquidity Spiral.

SIMULATED. Stretch implementation of a multi-leg macro package with dynamic delta hedging
subject to Almgren-Chriss (2001) market impact and endogenous margin spiral constraints.
"""

from __future__ import annotations

import math

import numpy as np


def almgren_chriss_impact(shares: float, adv: float, volatility: float, eta: float = 0.1) -> float:
    """Temporary price impact cost under Almgren-Chriss power-law model."""
    participation = abs(shares) / adv
    return float(eta * volatility * (participation**0.6) * abs(shares))


def main() -> None:
    rng = np.random.default_rng(42)
    n_steps = 20
    dt = 1.0 / 252.0

    # Macro asset: Equity Index Future
    spot = 100.0
    vol = 0.25
    adv = 1_000_000.0  # Daily volume in contracts

    # Initial position: Short 100 Calls (K=100, T=0.25) hedged dynamically with underlying futures
    k = 100.0
    t_rem = 0.25
    r = 0.045

    notional_options = 10_000  # 10,000 call contracts
    hedge_position = 0.0
    cumulative_impact_cost = 0.0

    print("=== Dynamic Cross-Asset Delta Rebalancing Simulation ===")
    print(f"{'Step':<6}{'Spot':<10}{'Option Delta':<14}{'Hedge Trade':<14}{'Impact Cost':<12}")
    print("-" * 56)

    for step in range(n_steps):
        # Simulate price jump
        z = float(rng.normal(0.0, 1.0))
        spot *= math.exp((r - 0.5 * vol**2) * dt + vol * math.sqrt(dt) * z)
        t_rem = max(1e-4, t_rem - dt)

        # Compute BSM Delta
        sqt = vol * math.sqrt(t_rem)
        d1 = (math.log(spot / k) + (r + 0.5 * vol**2) * t_rem) / sqt
        call_delta = 0.5 * (1.0 + math.erf(d1 / math.sqrt(2.0)))

        # Target hedge position
        target_hedge = notional_options * call_delta
        trade_shares = target_hedge - hedge_position

        # Market impact cost
        impact = almgren_chriss_impact(trade_shares, adv, vol)
        cumulative_impact_cost += impact
        hedge_position = target_hedge

        if step % 4 == 0:
            print(f"{step:<6}{spot:<10.2f}{call_delta:<14.4f}{trade_shares:<14.0f}${impact:<11.2f}")

    print("-" * 56)
    print(f"Final Hedge Position (Contracts): {hedge_position:,.0f}")
    print(f"Total Cumulative Execution Drag: ${cumulative_impact_cost:,.2f}")


main()
