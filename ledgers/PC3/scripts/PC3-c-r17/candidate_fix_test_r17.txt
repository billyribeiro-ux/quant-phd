def test_a_fill_takes_its_side_from_its_price_not_its_arrival() -> None:
    """Candidate fix (round 17): a fill at the bid is a purchase by the maker, whatever the arrival's name says."""
    turns = [turn_record(1, bid=9_990, ask=10_010, filled=True, arrival="uninformed-buy", price=9_990, inventory=1, cash=-9_990)]
    turns += [turn_record(t, size=0, inventory=1, cash=-9_990) for t in range(2, HORIZON + 1)]
    log = hand_log(turns, 10_000)
    expect(sol.replay(log) == oracle(log), "a fill at the bid is a purchase by the maker, whatever its arrival name")
