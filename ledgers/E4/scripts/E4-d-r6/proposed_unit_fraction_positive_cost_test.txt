# Proposed addition to course/modules/E/E4/grader/test_project.py (verifier round 6; NOT applied to the course tree).
# Covers the stated behaviour "indifference_gap is the gap of Proposition 3 at lambda*" and the informativeness at lambda* = 1
# for a positive information cost. The existing unit-fraction test uses cost 0, where exp(a c) = 1 and exp(-c) is not tested.
# Input check: tau_bar(0.05) = 2 exp(-0.2) - 1 = 0.6375 >= tau_y(1) = 1 / (1 + 4 * 0.6^2) = 0.4098, so lambda* = 1 (Prop 4).

def test_unit_fraction_is_reported_with_a_positive_cost() -> None:
    out = run(0.6, 0.05)
    expect(
        out["informed_fraction"] == 1.0,
        "the informed fraction must be 1 when the gap at 1 is nonnegative: check the boundary rule",
    )
    expect(
        abs(out["price_informativeness"] - oracle_informativeness(1.0, 0.6)) <= TOL_MOMENT,
        "the informativeness at fraction 1 should be the squared correlation of the payoff and the price, whatever the cost",
    )
    expect(
        abs(out["indifference_gap"] - oracle_gap(1.0, 0.6, 0.05)) <= TOL_GAP,
        "the reported gap at fraction 1 must be the informed-minus-uninformed expected utility there, net of the cost",
    )
