
def test_large_information_cost_is_accepted() -> None:
    """A finite cost above the named range is valid: the table names only negative costs. At cost 2 the gap at 0 is
    negative, so the boundary rule gives fraction 0, and the reported gap is the limit at 0."""
    out = run(0.6, 2.0)
    expect(
        out["informed_fraction"] == 0.0,
        "the informed fraction must be 0 when the gap at 0 is nonpositive: check the boundary rule at a high cost",
    )
    expect(
        abs(out["indifference_gap"] - oracle_gap(0.0, 0.6, 2.0)) <= TOL_GAP,
        "the reported gap at fraction 0 must be the limit of the gap as the fraction tends to 0",
    )
